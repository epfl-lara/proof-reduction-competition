theorem putnam_1995_a3
(relation : (Fin 9 → ℤ) → (Fin 9 → ℤ) → Prop)
(digits_to_num : (Fin 9 → ℤ) → ℤ)
(hdigits_to_num : digits_to_num = fun dig => ∑ i : Fin 9, (dig i) * 10^i.1)
(hrelation : ∀ d e : (Fin 9 → ℤ), relation d e ↔ (∀ i : Fin 9, d i < 10 ∧ d i ≥ 0 ∧ e i < 10 ∧ e i ≥ 0) ∧ (∀ i : Fin 9, 7 ∣ (digits_to_num (fun j : Fin 9 => if j = i then e j else d j))))

      7 ∣ digits_to_num a + (b i - a i) * 10^i.1 := by

        digits_to_num a + (b i - a i) * 10^i.1 := by
      rw [hdigits_to_num]
      calc
        _ = ∑ j : Fin 9, (a j * 10^j.1 +
            if j = i then (b j - a j) * 10^j.1 else 0) := by
          apply Finset.sum_congr rfl
          intro j _
          by_cases hij : j = i <;> simp [hij] <;> ring
        _ = _ := by simp [Finset.sum_add_distrib]
    rw [← eqn]
    exact ((hrelation a b).mp h).2 i
  intro d e f h i
  have hs := Finset.dvd_sum (s := Finset.univ) (fun j _ => step d e h.1 j)

    simp [hdigits_to_num]
    ring

    convert dvd_sub hs (dvd_mul_right (7 : ℤ) (digits_to_num d)) using 1 <;> ring
