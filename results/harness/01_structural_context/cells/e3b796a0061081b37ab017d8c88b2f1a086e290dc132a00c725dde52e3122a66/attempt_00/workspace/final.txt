open OldExpressions in
theorem extractedOldExprInVars :
  NormalizedOldExpr post →
  (extractOldExprVars post).Subset
  (Imperative.HasVarsPure.getVars post) := by
  have app {a b c d : List Expression.Ident}
      (h : a.Subset c) (k : b.Subset d) : (a ++ b).Subset (c ++ d) := by
    intro x hx
    simp only [List.mem_append] at hx ⊢
    exact hx.elim (fun hx => Or.inl (h hx)) (fun hx => Or.inr (k hx))
  intro h
  induction h <;>
    simp only [Imperative.HasVarsPure.getVars, extractOldExprVars,
      Lambda.LExpr.LExpr.getVars] at *
  all_goals try first
    | exact fun _ h => h
    | exact List.nil_subset _
    | assumption
    | exact app ‹_› ‹_›
    | exact app (app ‹_› ‹_›) ‹_›
  rename_i hn he hold ih₁ ih₂
  unfold extractOldExprVars
  split
  · simp [Lambda.LExpr.LExpr.getVars]
    exact fun _ h => h
  · next hf =>
      have hv := hold IsOldPred.oldPred
      cases hv
      exfalso
      apply hf
      rfl
  · exact app ih₁ ih₂
