theorem fiberwise_dist_lt_imp_dist_lt_unique_decoding_radius (i : Fin ℓ) (steps : ℕ)
    [NeZero steps] (h_i_add_steps : i.val + steps ≤ ℓ)
    (f : OracleFunction 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) ⟨i, by omega⟩)
  (h_fw_dist_lt : fiberwiseClose 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate)
    (i := i) (steps := steps) (h_i_add_steps := h_i_add_steps) (f := f)) :
  hammingClose 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) ⟨i, by omega⟩ f := by
  obtain ⟨g, _, hg⟩ : ∃ g : BBF_Code 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) i.castSucc, g ∈ Set.univ ∧
      (fiberwiseDisagreementSet 𝔽q β i steps h_i_add_steps f g.val).ncard =
        fiberwiseDistance 𝔽q β i steps h_i_add_steps f :=
    Nat.sInf_mem (Set.image_nonempty.mpr Set.univ_nonempty)
  let Y := fiberwiseDisagreementSet 𝔽q β i steps h_i_add_steps f g.val
  let hY := Y.toFinite
  let q := iteratedQuotientMap 𝔽q β h_ℓ_add_R_rate i steps h_i_add_steps
  let F := qMap_total_fiber 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) (i := ⟨i, lt_trans i.isLt (Nat.lt_of_add_right_lt h_ℓ_add_R_rate)⟩) steps
    (fin_ℓ_steps_lt_ℓ_add_R i steps h_i_add_steps)
  have hc : hammingDist f g.val ≤ Y.ncard * 2 ^ steps := by
    calc
      _ ≤ (hY.toFinset ×ˢ (Finset.univ : Finset (Fin (2 ^ steps)))).card := by
        apply Finset.card_le_card_of_surjOn (fun p => F p.1 p.2)
        intro x hx
        refine ⟨(q x, pointToIterateQuotientIndex 𝔽q β
          i.castSucc steps h_i_add_steps x), ?_, ?_⟩
        · exact Finset.mem_product.mpr ⟨hY.mem_toFinset.mpr
            ⟨x, rfl, (Finset.mem_filter.mp hx).2⟩, Finset.mem_univ _⟩
        · exact (is_fiber_iff_generates_quotient_point 𝔽q β i steps h_i_add_steps
            x (q x)).mp rfl
      _ = _ := by
        simp only [Finset.card_product, Finset.card_univ, Fintype.card_fin,
          ← Y.ncard_eq_toFinset_card hY]
  unfold hammingClose
  apply lt_of_le_of_lt (mul_le_mul_left'
    (Code.distFromCode_le_dist_to_mem f g.val g.property) 2)
  apply WithTop.coe_lt_coe.mpr
  apply Nat.lt_succ_of_le
  have h := Nat.mul_le_mul_right (2 ^ steps)
    (Nat.le_of_lt_succ (WithTop.coe_lt_coe.mp h_fw_dist_lt))
  simp only [← hg, Nat.mul_assoc, Nat.sub_mul, ← Nat.pow_add,
    Nat.sub_add_eq,
    Nat.sub_add_cancel (Nat.le_sub_of_add_le'
      (h_i_add_steps.trans (Nat.le_add_right ℓ 𝓡))),
    Nat.sub_add_cancel (Nat.le_sub_of_add_le' h_i_add_steps)] at h
  exact (Nat.mul_le_mul_left 2 hc).trans h
