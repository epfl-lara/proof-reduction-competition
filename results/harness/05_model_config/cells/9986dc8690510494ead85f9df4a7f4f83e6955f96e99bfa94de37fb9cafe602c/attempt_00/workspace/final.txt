lemma Typing.progress (der : Typing [] t τ) : t.Value ∨ ∃ t', t ⭢βᵛ t' := by
  generalize eq : [] = Γ at der
  let der' := der
  induction der <;> subst eq
  case var mem => cases mem
  case abs | tabs => left; constructor; exact der'.wf.2.1
  case app t₁ _ _ t₂ l r ih_l ih_r =>
    specialize ih_l rfl
    specialize ih_r rfl
    grind only [→ l.canonical_form_abs, → wf, → Red.abs,
      → Red.appₗ (t₁ := t₁), → Red.appᵣ (t₂ := t₂)]
  case tapp d _ ih =>
    cases der'.wf.2.1 with | tapp _ lc =>
      specialize ih rfl
      grind only [→ d.canonical_form_tabs, → Red.tabs, → Red.tapp]
  case let' ih _ =>
    specialize ih rfl
    grind only [body_let.mp der'.wf.2.1, → Red.let_body, → Red.let_bind]
  case inl | inr | sub =>
    rename_i ih
    specialize ih rfl
    grind only [Value.inl, Value.inr, → Red.inl, → Red.inr]
  case case t₁ _ _ _ _ _ _ d _ _ ih _ _ =>
    specialize ih rfl
    grind only [body_case.mp der'.wf.2.1, → d.canonical_form_sum, cases Value, → Red.case (t₁ := t₁),
      → Red.case_inl, → Red.case_inr]
