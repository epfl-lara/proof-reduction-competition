(u : ℕ → ℤ)

: (∃ N c : ℕ, c > 0 ∧ ∀ n ≥ N, u (n + c) = u n) := by

  obtain ⟨i, j, hij, he⟩ := Set.finite_univ.exists_lt_map_eq_of_forall_mem
    (f := f) (fun _ => Set.mem_univ _)
  have hr n := (hu (n + 4) (Nat.le_add_left _ _)).1
  have he' : ∀ m, f (i + m) = f (j + m) := Nat.rec he fun m ih => by

      simp only [ih, true_and]

  intro n hn
