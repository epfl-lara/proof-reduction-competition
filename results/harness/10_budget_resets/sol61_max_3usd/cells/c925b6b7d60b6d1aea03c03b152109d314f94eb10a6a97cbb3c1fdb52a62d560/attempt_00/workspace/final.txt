theorem fold_advances_evaluation_poly
  (i : Fin (ℓ)) (h_i_succ_lt : i + 1 < ℓ + 𝓡)
  (coeffs : Fin (2 ^ (ℓ - ↑i)) → L) (r_chal : L) :
  let P_i : L[X] := intermediateEvaluationPoly 𝔽q β h_ℓ_add_R_rate (i := ⟨i, by
    exact Nat.lt_trans (n := i) (k := ℓ+1) (m := ℓ) (h₁ := i.isLt) (by exact Nat.lt_add_one ℓ)
  ⟩) coeffs
  let f_i := fun (x : (sDomain 𝔽q β h_ℓ_add_R_rate)
      ⟨i, by exact Nat.lt_trans (n := i) (k := r) (m := ℓ) (h₁ := by omega) (by omega)⟩) =>
    P_i.eval (x.val : L)
  let f_i_plus_1 := fold (i := ⟨i, by omega⟩) (h_i := by omega) (f := f_i) (r_chal := r_chal)
  let new_coeffs := fun j : Fin (2^(ℓ - (i + 1))) =>
    (1 - r_chal) * (coeffs ⟨j.val * 2, by
      rw [←Nat.add_zero (j.val * 2)]
      apply mul_two_add_bit_lt_two_pow (c := ℓ - i) (a := j) (b := ℓ - (↑i + 1))
        (i := 0) (by omega) (by omega)
    ⟩) +
    r_chal * (coeffs ⟨j.val * 2 + 1, by
      apply mul_two_add_bit_lt_two_pow (c := ℓ - i) (a := j) (b := ℓ - (↑i + 1))
        (i := 1) (by omega) (by omega)
    ⟩)
  let P_i_plus_1 :=
    intermediateEvaluationPoly 𝔽q β h_ℓ_add_R_rate (i := ⟨i+1, by omega⟩) new_coeffs
  ∀ (y : (sDomain 𝔽q β h_ℓ_add_R_rate)
    ⟨i+1, by omega⟩), f_i_plus_1 y = P_i_plus_1.eval y.val := by
  dsimp only
  intro y
  let x := qMap_total_fiber 𝔽q β (i := ⟨i, by omega⟩) (steps := 1)
    (h_i_add_steps := h_i_succ_lt) y
  have hq (k : Fin 2) : (qMap 𝔽q β ⟨i, by omega⟩).eval (x k).val = y.val := by
    have h := generates_quotient_point_if_is_fiber_of_y 𝔽q β i 1 i.isLt
      (x k) y ⟨k, rfl⟩
    rw [iteratedQuotientMap_k_eq_1_is_qMap 𝔽q β h_ℓ_add_R_rate] at h
    exact (congrArg Subtype.val h).symm
  have hd : (x 1).val - (x 0).val = 1 := by
    simp only [x, qMap_total_fiber_one_level_eq 𝔽q β i i.isLt y,
      Fin2ToF2, Fin.isValue, one_ne_zero, ↓reduceIte, zero_smul, one_smul,
      zero_add, Submodule.coe_add, get_sDomain_first_basis_eq_1, add_sub_cancel_right]
  let A := evenRefinement 𝔽q β h_ℓ_add_R_rate i coeffs
  let B := oddRefinement 𝔽q β h_ℓ_add_R_rate i coeffs
  calc
    _ = (1 - r_chal) * A.eval y.val + r_chal * B.eval y.val := by
      simp only [fold, evaluation_poly_split_identity 𝔽q β h_ℓ_add_R_rate i coeffs,
        Polynomial.eval_add, eval_comp, Polynomial.eval_mul, Polynomial.eval_X,
        A, B]
      rw [hq 0, hq 1, sub_eq_iff_eq_add.mp hd]
      ring
    _ = _ := by
      simp only [A, B, evenRefinement, oddRefinement, intermediateEvaluationPoly, Fin.eta]
      rw! (castMode := .all) [Nat.sub_add_eq]
      simp only [Polynomial.eval_finset_sum, Polynomial.eval_mul, Polynomial.eval_C, mul_sum,
        ←sum_add_distrib, add_mul, mul_assoc, Fin.eta]
