lemma fundamental_theorem_of_variational_calculus' {f : Y → V}
    (μ : Measure Y) [IsFiniteMeasureOnCompacts μ] [μ.IsOpenPosMeasure]
    [OpensMeasurableSpace Y]
    (hf : Continuous f) (hg : ∀ g, IsTestFunction g → ∫ x, ⟪f x, g x⟫_ℝ ∂μ = 0) :
    f = 0 := by
  funext x
  apply (inner_self_eq_zero' (𝕜 := ℝ)).mp
  by_contra hx
  have hp := real_inner_self_nonneg'.lt_of_ne' hx
  have hc := Continuous.inner' (𝕜 := ℝ) f (fun _ => f x) hf continuous_const
  obtain ⟨φ, hs, hcomp, hsmooth, hnonneg, hxone⟩ :=
    exists_contDiff_tsupport_subset (n := ⊤) (hc.continuousAt.eventually_const_lt hp)
  have hz := hg _ (IsTestFunction.smul_right (g := fun _ => f x) ⟨hsmooth, hcomp⟩ contDiff_const)
  simp only [inner_smul_right'] at hz
  refine (ne_of_gt ?_) hz
  refine (hsmooth.continuous.mul hc).integral_pos_of_hasCompactSupport_nonneg_nonzero
    hcomp.mul_right ?_ (mul_ne_zero (hxone.trans_ne one_ne_zero) hx)
  intro y
  rcases (hnonneg ⟨y, rfl⟩).1.eq_or_lt with hy | hy
  · simp [← hy]
  · exact (mul_pos hy (hs (subset_tsupport φ hy.ne'))).le
