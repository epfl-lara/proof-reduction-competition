theorem parallelReduction_diamond : Diamond ParallelReduction := by
  intro a a₁ a₂ h₁ h₂
  induction h₁ generalizing a₂
  case refl => exact ⟨_, h₂, .refl _⟩
  all_goals cases h₂
  case par.par iha ihb a'' b'' ha'' hb'' =>
    obtain ⟨a₃, ha₁, ha₂⟩ := iha ha''
    obtain ⟨b₃, hb₁, hb₂⟩ := ihb hb''
    exact ⟨_, .par ha₁ hb₁, .par ha₂ hb₂⟩
  case' red_I.par | red_K.par | red_S.par => apply symm
  case par.red_I | red_I.par =>
    cases I_irreducible _ (by assumption)
    exact ⟨_, .red_I _, by assumption⟩
  case par.red_K | red_K.par =>
    obtain ⟨x, h, rfl⟩ := Ka_irreducible _ _ (by assumption)
    exact ⟨_, .red_K .., h⟩
  case par.red_S | red_S.par =>
    obtain ⟨x, y, hx, hy, rfl⟩ := Sab_irreducible _ _ _ (by assumption)
    refine ⟨_, .red_S .., ?_⟩
    apply ParallelReduction.par <;> apply ParallelReduction.par <;> assumption
  all_goals
    refine ⟨_, .refl _, ?_⟩
    constructor <;> assumption
