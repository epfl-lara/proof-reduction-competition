theorem bisimilarity_congr_choice :
    (p ~[lts (defs := defs)] q) → (choice p r) ~[lts (defs := defs)] (choice q r) := by
  intro h
  exists _, ChoiceBisim.choice h
  rintro _ _ (h | h) μ
  all_goals constructor <;> intro s' ht
  case' choice.left | choice.right =>
    cases ht
    case choiceR ht => use s', ht.choiceR, ChoiceBisim.bisim (HomBisimilarity.refl _)
  all_goals
    refine Exists.imp (fun _ => And.imp (by first | exact Tr.choiceL | exact id)
      ChoiceBisim.bisim) ?_
    first | apply h.follow_fst | apply h.follow_snd
    assumption
