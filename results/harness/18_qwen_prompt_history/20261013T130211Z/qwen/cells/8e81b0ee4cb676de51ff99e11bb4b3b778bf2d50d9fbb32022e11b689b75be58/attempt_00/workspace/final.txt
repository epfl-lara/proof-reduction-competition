lemma Typing.progress (der : Typing [] t τ) : t.Value ∨ ∃ t', t ⭢βᵛ t' := by
  generalize eq : [] = Γ at der
  induction der <;> subst eq
  case var mem => cases mem
  case' app l r ihl ihr =>
    cases ihl rfl with
    | inr red => exact .inr ⟨_, .appₗ r.wf.2.1 red.choose_spec⟩
    | inl vl =>
        cases ihr rfl with
        | inr red => exact .inr ⟨_, .appᵣ vl red.choose_spec⟩
        | inl vr =>
            obtain ⟨_, _, rfl⟩ := canonical_form_abs vl l
            exact .inr ⟨_, .abs vl.lc vr⟩
  case' tapp der sub ih =>
    cases ih rfl with
    | inr red => exact .inr ⟨_, .tapp sub.wf.2.1.lc red.choose_spec⟩
    | inl val =>
        obtain ⟨_, _, rfl⟩ := canonical_form_tabs val der
        exact .inr ⟨_, .tabs val.lc sub.wf.2.1.lc⟩
  case' let' L der h ih _ =>
    have ⟨_, b⟩ := body_let.mp (Typing.let' L der h).wf.2.1
    cases ih rfl with
    | inr red => exact .inr ⟨_, .let_bind red.choose_spec b⟩
    | inl val => exact .inr ⟨_, .let_body val b⟩
  case' inl _ _ ih => exact (ih rfl).imp Value.inl fun red => ⟨_, .inl red.choose_spec⟩
  case' inr _ _ ih => exact (ih rfl).imp Value.inr fun red => ⟨_, .inr red.choose_spec⟩
  case' case L der h₂ h₃ ih _ _ =>
    have ⟨_, b₂, b₃⟩ := body_case.mp (Typing.case L der h₂ h₃).wf.2.1
    cases ih rfl with
    | inr red => exact .inr ⟨_, .case red.choose_spec b₂ b₃⟩
    | inl val =>
        obtain ⟨_, rfl|rfl⟩ := canonical_form_sum val der
        · cases val with | inl val => exact .inr ⟨_, .case_inl val b₂ b₃⟩
        · cases val with | inr val => exact .inr ⟨_, .case_inr val b₂ b₃⟩
  case' sub _ _ ih => exact ih rfl
  case' abs L h _ => exact .inl (.abs ((Typing.abs L h).wf).2.1)
  case' tabs L h _ => exact .inl (.tabs ((Typing.tabs L h).wf).2.1)
