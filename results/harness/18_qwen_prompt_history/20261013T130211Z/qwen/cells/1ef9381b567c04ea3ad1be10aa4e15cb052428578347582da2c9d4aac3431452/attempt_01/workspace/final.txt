theorem putnam_1995_a3
(relation : (Fin 9 → ℤ) → (Fin 9 → ℤ) → Prop)
(digits_to_num : (Fin 9 → ℤ) → ℤ)
(hdigits_to_num : digits_to_num = fun dig => ∑ i : Fin 9, (dig i) * 10^i.1)
(hrelation : ∀ d e : (Fin 9 → ℤ), relation d e ↔ (∀ i : Fin 9, d i < 10 ∧ d i ≥ 0 ∧ e i < 10 ∧ e i ≥ 0) ∧ (∀ i : Fin 9, 7 ∣ (digits_to_num (fun j : Fin 9 => if j = i then e j else d j))))
: ∀ d e f : (Fin 9 → ℤ), ((relation d e) ∧ (relation e f)) → (∀ i : Fin 9, 7 ∣ d i - f i) := by
  intro d e f ⟨hde, hef⟩
  have key : ∀ (x y : Fin 9 → ℤ), relation x y → ∀ i : Fin 9,
      7 ∣ digits_to_num x + (y i - x i) * 10 ^ i.1 := by
    intro x y h i
    have hy := (hrelation x y).mp h |>.2 i
    rw [show (fun j => if j = i then y j else x j) = fun j => x j + (if j = i then y j - x j else 0) from by
      funext j
      by_cases e : j = i <;> simp [e]] at hy
    simpa only [hdigits_to_num, add_mul, ite_mul, zero_mul, Finset.sum_add_distrib,
      Finset.sum_ite_eq', Finset.mem_univ, ↓reduceIte] using hy
  have hs : ∑ i : Fin 9, (digits_to_num d + (e i - d i) * 10 ^ i.1)
      = digits_to_num d + digits_to_num e + 7 * digits_to_num d := by
    rw [Finset.sum_add_distrib, Finset.sum_const, Finset.card_univ, Fintype.card_fin]
    simp only [hdigits_to_num, sub_mul, Finset.sum_sub_distrib, Int.nsmul_eq_mul]
    ring
  have hsum : 7 ∣ digits_to_num d + digits_to_num e :=
    (dvd_add_left (dvd_mul_right 7 (digits_to_num d))).mp (hs ▸ Finset.dvd_sum fun i _ => key d e hde i)
  intro i
  exact (IsCoprime.pow_right (n := i.1) (by decide : IsCoprime (7 : ℤ) (10 : ℤ))).dvd_of_dvd_mul_right
    ((dvd_neg.mpr (dvd_add (dvd_sub (key e f hef i) hsum) (key d e hde i))).trans (dvd_of_eq (by ring)))
