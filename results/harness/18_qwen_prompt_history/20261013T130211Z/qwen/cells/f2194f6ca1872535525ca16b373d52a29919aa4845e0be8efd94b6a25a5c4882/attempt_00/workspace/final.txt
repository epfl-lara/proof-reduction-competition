theorem fold_advances_evaluation_poly
  (i : Fin (ℓ)) (h_i_succ_lt : i + 1 < ℓ + 𝓡)
  (coeffs : Fin (2 ^ (ℓ - ↑i)) → L) (r_chal : L) :
  let P_i : L[X] := intermediateEvaluationPoly 𝔽q β h_ℓ_add_R_rate (i := ⟨i, by
    exact Nat.lt_trans (n := i) (k := ℓ+1) (m := ℓ) (h₁ := i.isLt) (by exact Nat.lt_add_one ℓ)
  ⟩) coeffs
  let f_i := fun (x : (sDomain 𝔽q β h_ℓ_add_R_rate)
      ⟨i, by exact Nat.lt_trans (n := i) (k := r) (m := ℓ) (h₁ := by omega) (by omega)⟩) =>
    P_i.eval (x.val : L)
  let f_i_plus_1 := fold (i := ⟨i, by omega⟩) (h_i := by omega) (f := f_i) (r_chal := r_chal)
  let new_coeffs := fun j : Fin (2^(ℓ - (i + 1))) =>
    (1 - r_chal) * (coeffs ⟨j.val * 2, by
      rw [←Nat.add_zero (j.val * 2)]
      apply mul_two_add_bit_lt_two_pow (c := ℓ - i) (a := j) (b := ℓ - (↑i + 1))
        (i := 0) (by omega) (by omega)
    ⟩) +
    r_chal * (coeffs ⟨j.val * 2 + 1, by
      apply mul_two_add_bit_lt_two_pow (c := ℓ - i) (a := j) (b := ℓ - (↑i + 1))
        (i := 1) (by omega) (by omega)
    ⟩)
  let P_i_plus_1 :=
    intermediateEvaluationPoly 𝔽q β h_ℓ_add_R_rate (i := ⟨i+1, by omega⟩) new_coeffs
  ∀ (y : (sDomain 𝔽q β h_ℓ_add_R_rate)
    ⟨i+1, by omega⟩), f_i_plus_1 y = P_i_plus_1.eval y.val := by
  intro P_i f_i f_i_plus_1 new_coeffs P_i_plus_1 y
  let fiberMap := qMap_total_fiber 𝔽q β ⟨i, by omega⟩ 1
    (by simp only; omega) y
  let x₀ := fiberMap 0
  let x₁ := fiberMap 1
  let P₀ := evenRefinement 𝔽q β h_ℓ_add_R_rate i coeffs
  let P₁ := oddRefinement 𝔽q β h_ℓ_add_R_rate i coeffs
  have hq (k : Fin 2) : (AdditiveNTT.qMap 𝔽q β ⟨i, by omega⟩).eval (fiberMap k).val = y := by
    rw [generates_quotient_point_if_is_fiber_of_y 𝔽q β i 1 (by omega)
      (fiberMap k) y ⟨k, rfl⟩,
      iteratedQuotientMap_k_eq_1_is_qMap 𝔽q β h_ℓ_add_R_rate i (by omega) (fiberMap k)]
  have h_add : x₁.val = 1 + x₀.val := by
    have fz0 : Fin2ToF2 𝔽q 0 = 0 := rfl
    have fz1 : Fin2ToF2 𝔽q 1 = 1 := rfl
    have e₀ := qMap_total_fiber_one_level_eq 𝔽q β i (by omega) y 0
    have e₁ := qMap_total_fiber_one_level_eq 𝔽q β i (by omega) y 1
    have hb := get_sDomain_first_basis_eq_1 𝔽q β h_ℓ_add_R_rate ⟨i, by omega⟩ (fin_ℓ_lt_ℓ_add_R i)
    simp only [fz0, fz1, zero_smul, one_smul, x₀, x₁, fiberMap, Submodule.coe_add, e₀, e₁,
      zero_add, hb]
  have h_eval (k : Fin 2) :
      f_i (fiberMap k) = P₀.eval y.val + (fiberMap k).val * P₁.eval y.val := by
    simp only [evaluation_poly_split_identity 𝔽q β h_ℓ_add_R_rate i coeffs, f_i, P_i, P₀, P₁,
      Polynomial.eval_add, eval_comp, Polynomial.eval_mul, Polynomial.eval_X, hq k]
  have h_poly : Polynomial.C (1 - r_chal) * P₀ + Polynomial.C r_chal * P₁ = P_i_plus_1 := by
    simp only [P_i_plus_1, new_coeffs, intermediateEvaluationPoly, P₀, P₁, evenRefinement,
      oddRefinement, Finset.mul_sum, ← Finset.sum_add_distrib, Polynomial.C_add,
      Polynomial.C_mul, add_mul, ← mul_assoc]
    rfl
  calc f_i_plus_1 y
      = (P₀.eval y.val + x₀.val * P₁.eval y.val) * ((1 - r_chal) * x₁.val - r_chal) +
          (P₀.eval y.val + x₁.val * P₁.eval y.val) * (r_chal - (1 - r_chal) * x₀.val) := by
        simp only [f_i_plus_1, fold, x₀, x₁, fiberMap, h_eval 0, h_eval 1]
      _ = P_i_plus_1.eval y.val := by
        rw [h_add, ← h_poly]
        simp only [Polynomial.eval_add, Polynomial.eval_mul, Polynomial.eval_C]
        ring
