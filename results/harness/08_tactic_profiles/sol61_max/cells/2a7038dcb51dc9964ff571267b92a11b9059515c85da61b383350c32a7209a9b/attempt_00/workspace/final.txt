lemma fundamental_theorem_of_variational_calculus' {f : Y → V}
    (μ : Measure Y) [IsFiniteMeasureOnCompacts μ] [μ.IsOpenPosMeasure]
    [OpensMeasurableSpace Y]
    (hf : Continuous f) (hg : ∀ g, IsTestFunction g → ∫ x, ⟪f x, g x⟫_ℝ ∂μ = 0) :
    f = 0 := by
  funext x₀
  by_contra hx₀
  have hc := Continuous.inner' (𝕜 := ℝ) f (fun _ => f x₀) hf continuous_const
  have hpos := real_inner_self_nonneg'.lt_of_ne' (inner_self_eq_zero'.not.mpr hx₀)
  obtain ⟨φ, hφs, hφc, hφd, hφn, hφx⟩ :=
    exists_contDiff_tsupport_subset
      (hc.continuousAt.preimage_mem_nhds (Ioi_mem_nhds hpos))
  have hi := hg _
    (IsTestFunction.smul_right ⟨hφd, hφc⟩ (contDiff_const (c := f x₀)))
  simp only [inner_smul_right'] at hi
  refine ((hφd.continuous.mul hc).integral_pos_of_hasCompactSupport_nonneg_nonzero
    hφc.mul_right ?_ (mul_ne_zero (hφx.trans_ne one_ne_zero) hpos.ne')).ne' hi
  intro x
  by_cases hx : φ x = 0
  · simp [hx]
  · exact mul_nonneg (hφn ⟨x, rfl⟩).1
      (hφs (subset_closure hx)).le
