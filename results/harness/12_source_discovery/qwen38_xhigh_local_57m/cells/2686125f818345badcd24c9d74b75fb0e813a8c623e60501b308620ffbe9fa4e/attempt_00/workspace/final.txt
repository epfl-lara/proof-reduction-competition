lemma ι_timeOrderF_superCommuteF_eq_time {φ ψ : 𝓕.CrAnFieldOp}
    (hφψ : crAnTimeOrderRel φ ψ) (hψφ : crAnTimeOrderRel ψ φ) (a b : 𝓕.FieldOpFreeAlgebra) :
    ι 𝓣ᶠ(a * [ofCrAnOpF φ, ofCrAnOpF ψ]ₛF * b) =
    ι ([ofCrAnOpF φ, ofCrAnOpF ψ]ₛF * 𝓣ᶠ(a * b)) := by
  refine Submodule.span_induction₂ (R := ℂ) (s := Set.range ofCrAnListFBasis)
    (t := Set.range ofCrAnListFBasis)
    (p := fun x y _ _ => ι 𝓣ᶠ(x * [ofCrAnOpF φ, ofCrAnOpF ψ]ₛF * y) =
      ι ([ofCrAnOpF φ, ofCrAnOpF ψ]ₛF * 𝓣ᶠ(x * y)))
    ?mem ?zl ?zr ?al ?ar ?sl ?sr (Basis.mem_span _ a) (Basis.mem_span _ b)
  · intro x y hx hy
    obtain ⟨φs', rfl⟩ := hx
    obtain ⟨φs, rfl⟩ := hy
    simp only [ofListBasis_eq_ofList, map_mul]
    let k := 𝓢(𝓕.crAnStatistics φ, 𝓕.crAnStatistics ψ)
    let A := crAnTimeOrderSign (φs' ++ [φ, ψ] ++ φs)
    let l := List.insertionSort crAnTimeOrderRel (φs' ++ φs)
    let t := List.takeWhile (fun c => ¬ crAnTimeOrderRel φ c) l
    let f := List.filter (fun c => crAnTimeOrderRel φ c ∧ crAnTimeOrderRel c φ) φs'
    let g := List.filter (fun c => crAnTimeOrderRel φ c ∧ crAnTimeOrderRel c φ) φs
    let u := List.filter (fun c => crAnTimeOrderRel φ c ∧ ¬ crAnTimeOrderRel c φ) l
    calc _
      _ = crAnTimeOrderSign (φs' ++ φ :: ψ :: φs) •
          ι (ofCrAnListF (crAnTimeOrderList (φs' ++ φ :: ψ :: φs))) -
          crAnTimeOrderSign (φs' ++ ψ :: φ :: φs) • k •
          ι (ofCrAnListF (crAnTimeOrderList (φs' ++ ψ :: φ :: φs))) := by
        conv_lhs =>
          rw [← ofCrAnListF_singleton, ← ofCrAnListF_singleton,
            superCommuteF_ofCrAnListF_ofCrAnListF]
        simp [mul_sub, sub_mul, ← ofCrAnListF_append]
        rw [timeOrderF_ofCrAnListF, timeOrderF_ofCrAnListF]
        simp only [map_smul, sub_right_inj]
        module
      _ = crAnTimeOrderSign (φs' ++ φ :: ψ :: φs) •
          (ι (ofCrAnListF (crAnTimeOrderList (φs' ++ φ :: ψ :: φs))) - k •
            ι (ofCrAnListF (crAnTimeOrderList (φs' ++ ψ :: φ :: φs)))) := by
        rw [crAnTimeOrderSign_swap_eq_time hφψ hψφ]
        module
      _ = A • (ι (ofCrAnListF t) * ι (ofCrAnListF f) * ι (ofCrAnListF [φ, ψ]) *
            ι (ofCrAnListF g) * ι (ofCrAnListF u) -
            k • ι (ofCrAnListF t) * ι (ofCrAnListF f) * ι (ofCrAnListF [ψ, φ]) *
            ι (ofCrAnListF g) * ι (ofCrAnListF u)) := by
        have h1 := insertionSort_of_eq_list 𝓕.crAnTimeOrderRel φ φs' [φ, ψ] φs (by simp [hφψ, hψφ])
        rw [crAnTimeOrderList, show φs' ++ φ :: ψ :: φs = φs' ++ [φ, ψ] ++ φs by simp, h1]
        have h2 := insertionSort_of_eq_list 𝓕.crAnTimeOrderRel φ φs' [ψ, φ] φs (by simp [hφψ, hψφ])
        rw [crAnTimeOrderList, show φs' ++ ψ :: φ :: φs = φs' ++ [ψ, φ] ++ φs by simp, h2]
        repeat rw [ofCrAnListF_append]
        simp only [map_mul, Algebra.smul_mul_assoc]
        rfl
      _ = A • (ι (ofCrAnListF t) * ι (ofCrAnListF f) *
            (ι (ofCrAnListF [φ, ψ]) - k • ι (ofCrAnListF [ψ, φ])) *
            ι (ofCrAnListF g) * ι (ofCrAnListF u)) := by
        simp only [mul_sub, sub_mul, Algebra.mul_smul_comm, Algebra.smul_mul_assoc]
      _ = crAnTimeOrderSign (φs' ++ [φ, ψ] ++ φs) • (ι (ofCrAnListF t) * ι (ofCrAnListF f) *
          ι [ofCrAnOpF φ, ofCrAnOpF ψ]ₛF * ι (ofCrAnListF g) * ι (ofCrAnListF u)) := by
        congr
        rw [superCommuteF_ofCrAnOpF_ofCrAnOpF, ← ofCrAnListF_singleton, ← ofCrAnListF_singleton]
        simp only [← ofCrAnListF_append, List.cons_append, List.nil_append, Algebra.smul_mul_assoc,
          map_sub, map_smul]
        rfl
    rw [Subalgebra.mem_center_iff.mp (ι_superCommuteF_ofCrAnOpF_ofCrAnOpF_mem_center φ ψ)]
    repeat rw [mul_assoc]
    rw [← map_mul, ← map_mul, ← map_mul, ← ofCrAnListF_append, ← ofCrAnListF_append,
      ← ofCrAnListF_append]
    have h1 := insertionSort_of_takeWhile_filter 𝓕.crAnTimeOrderRel φ φs' φs
    simp only [ List.append_assoc, List.cons_append, List.nil_append] at h1 ⊢
    rw [← h1, ← crAnTimeOrderList]
    by_cases hq : (𝓕 |>ₛ φ) ≠ (𝓕 |>ₛ ψ)
    · simp only [ι_superCommuteF_of_diff_statistic hq, zero_mul, smul_zero]
    · rw [crAnTimeOrderSign, Wick.koszulSign_eq_rel_eq_stat _ _ hφψ hψφ (by grind only),
        ← crAnTimeOrderSign, ← ofCrAnListF_append, timeOrderF_ofCrAnListF]
      simp only [map_smul, Algebra.mul_smul_comm]
  · intros; simp only [zero_mul, mul_zero, map_zero]
  · intros; simp only [zero_mul, mul_zero, map_zero]
  · intros; simp only [add_mul, mul_add, map_add, *]
  · intros; simp only [add_mul, mul_add, map_add, *]
  · intros; simp only [Algebra.smul_mul_assoc, Algebra.mul_smul_comm, map_smul, *]
  · intros; simp only [Algebra.smul_mul_assoc, Algebra.mul_smul_comm, map_smul, *]
