theorem putnam_1995_a3
(relation : (Fin 9 → ℤ) → (Fin 9 → ℤ) → Prop)
(digits_to_num : (Fin 9 → ℤ) → ℤ)
(hdigits_to_num : digits_to_num = fun dig => ∑ i : Fin 9, (dig i) * 10^i.1)
(hrelation : ∀ d e : (Fin 9 → ℤ), relation d e ↔ (∀ i : Fin 9, d i < 10 ∧ d i ≥ 0 ∧ e i < 10 ∧ e i ≥ 0) ∧ (∀ i : Fin 9, 7 ∣ (digits_to_num (fun j : Fin 9 => if j = i then e j else d j))))
: ∀ d e f : (Fin 9 → ℤ), ((relation d e) ∧ (relation e f)) → (∀ i : Fin 9, 7 ∣ d i - f i) := by
  let w : Fin 9 → ℤ := fun j => (10 : ℤ) ^ j.1
  have patch (x y : Fin 9 → ℤ) (i : Fin 9) :
      digits_to_num (fun j => if j = i then y j else x j) =
        digits_to_num x + (y i - x i) * w i := by
    rw [hdigits_to_num]
    calc
      (∑ j : Fin 9, (if j = i then y j else x j) * w j) =
          ∑ j : Fin 9, (x j * w j + (if j = i then (y j - x j) else 0) * w j) := by
        apply Finset.sum_congr rfl
        intro j hj
        by_cases h : j = i
        · subst j
          simp only [w, if_true]
          ring
        · simp only [w, if_neg h, zero_mul, add_zero]
      _ = (∑ j : Fin 9, x j * w j) + (y i - x i) * w i := by
        rw [Finset.sum_add_distrib]
        simp [w, Finset.sum_ite_eq', Finset.mem_univ, i.is_lt]
  intro d e f h i

  have he : ∀ j : Fin 9, 7 ∣ (digits_to_num (fun k : Fin 9 => if k = j then f k else e k)) :=
    (hrelation e f).mp h.2 |>.2

    intro j
    have := he j

    simpa [add_comm] using this

      intro j hj
      exact ha j

      ring
