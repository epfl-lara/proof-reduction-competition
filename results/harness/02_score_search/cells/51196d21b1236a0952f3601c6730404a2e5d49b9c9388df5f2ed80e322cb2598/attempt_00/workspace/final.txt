theorem InitsUpdatesComm
  {P : PureExpr} {σ σ' σ'' : SemanticStore P}
  {ks ks' : List P.Ident} {vs vs' : List P.Expr} :
  UpdateStates σ ks vs σ' →
  InitStates σ' ks' vs' σ'' →
  ∃ σ₁,
    σ₁ = (updatedStates σ ks' vs') ∧
    InitStates σ ks' vs' σ₁ ∧
    UpdateStates σ₁ ks vs σ'' := by
  intro Hup Hinit
  refine ⟨updatedStates σ ks' vs', rfl, ?_, ?_⟩
  · exact updatedStatesInit (InitStatesLength Hinit)
      (UpdateStatesNotDefMonotone' (InitStatesNotDefined Hinit) Hup)
      (InitStatesNodup Hinit)
  · have hlenU := UpdateStatesLength Hup
    have hcomm := updatedStatesComm (σ:=σ) (kvs:=ks'.zip vs') (kvs':=ks.zip vs) (by
      rw [List.unzip_zip, List.unzip_zip]
      · intro k hk hk'
        simpa [InitStatesNotDefined Hinit k hk] using
          (UpdateStatesDefined Hup k hk')
      · exact hlenU
      · exact InitStatesLength Hinit)
    have hdef : isDefined (updatedStates σ ks' vs') ks :=
      updatedStatesDefMonotone (UpdateStatesDefined' Hup)
    simpa only [updatedStates, InitStatesUpdated Hinit, UpdateStatesUpdated Hup, hcomm] using
      (updatedStatesUpdate hlenU hdef)
