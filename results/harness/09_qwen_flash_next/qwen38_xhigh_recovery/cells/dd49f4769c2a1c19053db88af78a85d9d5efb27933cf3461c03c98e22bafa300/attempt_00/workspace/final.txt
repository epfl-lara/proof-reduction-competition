theorem putnam_1964_b2
(S : Type*) [Fintype S] [Nonempty S]
(P : Finset (Set S))
(hPP : ∀ T ∈ P, ∀ U ∈ P, T ∩ U ≠ ∅)
(hPS : ¬∃ T : Set S, T ∉ P ∧ (∀ U ∈ P, T ∩ U ≠ ∅))
: (P.card = 2 ^ (Fintype.card S - 1)) := by
  have := (Set.Intersecting.is_max_iff_card_eq fun a h b i => hPP a h b i ∘ Disjoint.inter_eq).mp fun t ht h => h.antisymm fun x hx => of_not_not (hPS ⟨x, ·, fun U g => ht hx (h g) ∘ Set.disjoint_iff_inter_eq_empty.mpr⟩)
  rw [Fintype.card_set, ← pow_sub_one_mul Fintype.card_ne_zero] at this
  omega
