theorem putnam_1995_a3
(relation : (Fin 9 → ℤ) → (Fin 9 → ℤ) → Prop)
(digits_to_num : (Fin 9 → ℤ) → ℤ)
(hdigits_to_num : digits_to_num = fun dig => ∑ i : Fin 9, (dig i) * 10^i.1)
(hrelation : ∀ d e : (Fin 9 → ℤ), relation d e ↔ (∀ i : Fin 9, d i < 10 ∧ d i ≥ 0 ∧ e i < 10 ∧ e i ≥ 0) ∧ (∀ i : Fin 9, 7 ∣ (digits_to_num (fun j : Fin 9 => if j = i then e j else d j))))
: ∀ d e f : (Fin 9 → ℤ), ((relation d e) ∧ (relation e f)) → (∀ i : Fin 9, 7 ∣ d i - f i) := by
  intro d e f h
  have hn := congrFun hdigits_to_num
  have change_digit (a b : Fin 9 → ℤ) (i : Fin 9) :
      digits_to_num (fun j => if j = i then b j else a j) =
        digits_to_num a + (b i * 10 ^ i.val - a i * 10 ^ i.val) := by
    rw [hdigits_to_num, ← sub_eq_iff_eq_add', ← Finset.sum_sub_distrib]
    simp only [ite_mul, ite_sub, sub_self, Fintype.sum_ite_eq']
  have hde := ((hrelation d e).mp h.1).2
  have hef := ((hrelation e f).mp h.2).2
  simp only [change_digit] at hde hef
  have hs := Finset.dvd_sum (s := Finset.univ) (fun i _ => hde i)
  simp only [Finset.sum_add_distrib, Finset.sum_const, Finset.card_univ,
    Fintype.card_fin, Int.nsmul_eq_mul,
    Finset.sum_sub_distrib, ← hn] at hs

  intro i
  apply hc.pow_right.dvd_of_dvd_mul_right
  refine (congrArg (7 ∣ ·) ?_).mp (dvd_sub (dvd_sub hs (dvd_mul_right 7 (digits_to_num d)))
      (dvd_add (hde i) (hef i)))
