theorem parallelReduction_diamond : Diamond ParallelReduction := by
  intro a a₁ a₂ h₁ h₂
  induction h₁ generalizing a₂
  case refl => exact ⟨_, h₂, .refl _⟩
  all_goals cases h₂
  case' par.red_I | par.red_K | par.red_S => refine _root_.symm ?_
  case red_I.par | par.red_I =>
    obtain rfl := I_irreducible _ ‹_›
    exact ⟨_, ‹_›, .red_I _⟩
  case red_K.par | par.red_K =>
    obtain ⟨x, hx, rfl⟩ := Ka_irreducible _ _ ‹_›
    exact ⟨_, hx, .red_K ..⟩
  case red_S.par | par.red_S =>
    obtain ⟨x, y, hx, hy, rfl⟩ := Sab_irreducible _ _ _ ‹_›
    exact ⟨_, .par (.par hx ‹_›) (.par hy ‹_›), .red_S ..⟩
  case par.par =>
    rename_i iha ihb e f ha' hb'
    obtain ⟨x, hx⟩ := iha ha'
    obtain ⟨y, hy⟩ := ihb hb'
    exact ⟨_, .par hx.1 hy.1, .par hx.2 hy.2⟩
  all_goals exact ⟨_, .refl _, by constructor <;> assumption⟩
