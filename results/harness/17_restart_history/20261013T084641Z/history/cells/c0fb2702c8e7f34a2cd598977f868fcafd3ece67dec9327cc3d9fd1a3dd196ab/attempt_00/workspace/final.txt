theorem putnam_1964_b2
(S : Type*) [Fintype S] [Nonempty S]
(P : Finset (Set S))
(hPP : ∀ T ∈ P, ∀ U ∈ P, T ∩ U ≠ ∅)
(hPS : ¬∃ T : Set S, T ∉ P ∧ (∀ U ∈ P, T ∩ U ≠ ∅))
: (P.card = 2 ^ (Fintype.card S - 1)) := by
  simp only [Ne, ← Set.disjoint_iff_inter_eq_empty] at hPP hPS
  apply Nat.mul_left_cancel Nat.two_pos
  rw [mul_pow_sub_one Fintype.card_ne_zero, ← Fintype.card_set]
  apply (Set.Intersecting.is_max_iff_card_eq hPP).mp
  intro Q hQ hPQ
  apply Finset.Subset.antisymm hPQ
  intro A hA
  by_contra h
  exact hPS ⟨A, h, fun U hU => hQ hA (hPQ hU)⟩
