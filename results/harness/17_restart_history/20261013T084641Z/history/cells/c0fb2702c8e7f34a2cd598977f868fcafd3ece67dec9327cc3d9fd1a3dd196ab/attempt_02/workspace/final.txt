theorem putnam_1964_b2
(S : Type*) [Fintype S] [Nonempty S]
(P : Finset (Set S))
(hPP : ∀ T ∈ P, ∀ U ∈ P, T ∩ U ≠ ∅)
(hPS : ¬∃ T : Set S, T ∉ P ∧ (∀ U ∈ P, T ∩ U ≠ ∅))
: (P.card = 2 ^ (Fintype.card S - 1)) := by
  simp only [Set.disjoint_iff_inter_eq_empty.not.symm, not_exists, and_comm.not.trans not_and_not_right] at *
  apply Nat.mul_left_cancel Nat.two_pos
  rw [mul_pow_sub_one Fintype.card_ne_zero, Fintype.card_set.symm,
    ← Set.Intersecting.is_max_iff_card_eq hPP]
  intro Q hQ hPQ
  apply hPQ.antisymm
  intro A hA
  apply hPS
  exact Set.Subset.trans hPQ (hQ hA)
