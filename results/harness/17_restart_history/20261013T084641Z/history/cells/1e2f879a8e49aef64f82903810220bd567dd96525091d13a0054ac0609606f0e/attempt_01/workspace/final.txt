theorem bisimilarity_congr_choice :
    (p ~[lts (defs := defs)] q) → (choice p r) ~[lts (defs := defs)] (choice q r) := by
  intro h
  exists ChoiceBisim, ChoiceBisim.choice h
  intro s t hr μ
  cases hr <;> rename_i h
  all_goals constructor <;> intro s' ht
  case' choice.left | choice.right =>
    cases ht <;> rename_i ht
  case choice.left.choiceR | choice.right.choiceR =>
    exact ⟨_, .choiceR ht, .bisim (HomBisimilarity.refl _)⟩
  all_goals
    first | apply (h.follow_fst ht).elim | apply (h.follow_snd ht).elim
    rintro t' ⟨ht', h'⟩
    refine ⟨t', ?_, .bisim h'⟩
    first | assumption | exact .choiceL ht'
