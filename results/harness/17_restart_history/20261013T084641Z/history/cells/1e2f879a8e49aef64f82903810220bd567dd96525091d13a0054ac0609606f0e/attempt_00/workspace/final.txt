theorem bisimilarity_congr_choice :
    (p ~[lts (defs := defs)] q) → (choice p r) ~[lts (defs := defs)] (choice q r) := by
  intro h
  refine ⟨_, ChoiceBisim.choice h, ?_⟩
  intro s t hr μ
  cases hr with
  | choice h =>
    constructor <;> intro s' ht <;> cases ht with
    | choiceL ht =>
      first | apply (h.follow_fst ht).elim | apply (h.follow_snd ht).elim
      rintro t' ⟨ht', h'⟩
      exact ⟨t', .choiceL ht', .bisim h'⟩
    | choiceR ht =>
      exact ⟨_, .choiceR ht, .bisim (HomBisimilarity.refl _)⟩
  | bisim h =>
    grind only [→ h.follow_fst, → h.follow_snd, → ChoiceBisim.bisim]
