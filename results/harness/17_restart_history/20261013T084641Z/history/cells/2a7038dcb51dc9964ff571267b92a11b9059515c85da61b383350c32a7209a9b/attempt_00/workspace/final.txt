lemma fundamental_theorem_of_variational_calculus' {f : Y → V}
    (μ : Measure Y) [IsFiniteMeasureOnCompacts μ] [μ.IsOpenPosMeasure]
    [OpensMeasurableSpace Y]
    (hf : Continuous f) (hg : ∀ g, IsTestFunction g → ∫ x, ⟪f x, g x⟫_ℝ ∂μ = 0) :
    f = 0 := by
  ext x
  by_contra hx
  have hp : 0 < ⟪f x, f x⟫_ℝ :=
    real_inner_self_nonneg'.lt_of_ne' (mt inner_self_eq_zero'.mp hx)
  have hc := Continuous.inner' (𝕜 := ℝ) f (fun _ => f x) hf continuous_const
  obtain ⟨φ, hsub, hcomp, hsmooth, hrange, heq⟩ :=
    exists_contDiff_tsupport_subset (hc.continuousAt (Ioi_mem_nhds hp))
  have hz := hg (fun y => φ y • f x)
    (IsTestFunction.smul_right ⟨hsmooth, hcomp⟩ contDiff_const)
  simp only [inner_smul_right'] at hz
  refine ((hsmooth.continuous.mul hc).integral_pos_of_hasCompactSupport_nonneg_nonzero (x := x)
    hcomp.mul_right ?_ ?_).ne' hz
  · intro y
    by_cases hy : φ y = 0
    · simp [hy]
    · exact mul_nonneg (hrange ⟨y, rfl⟩).1
        (hsub (subset_tsupport φ hy)).le
  · simpa [heq] using hp.ne'
