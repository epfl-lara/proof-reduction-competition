theorem fiberwise_dist_lt_imp_dist_lt_unique_decoding_radius (i : Fin ℓ) (steps : ℕ)
    [NeZero steps] (h_i_add_steps : i.val + steps ≤ ℓ)
    (f : OracleFunction 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) ⟨i, by omega⟩)
  (h_fw_dist_lt : fiberwiseClose 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate)
    (i := i) (steps := steps) (h_i_add_steps := h_i_add_steps) (f := f)) :
  hammingClose 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) ⟨i, by omega⟩ f := by
  let C := BBF_Code 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) ⟨i, fin_ℓ_lt_ℓ_add_one i⟩
  let Y (g : C) := fiberwiseDisagreementSet 𝔽q β i steps h_i_add_steps f g.val
  let S := (fun g => (Y g).ncard) '' Set.univ
  obtain ⟨g, _, hg⟩ := Nat.sInf_mem (Set.image_nonempty.mpr Set.univ_nonempty : S.Nonempty)
  change (Y g).ncard = fiberwiseDistance 𝔽q β i steps h_i_add_steps f at hg
  let q := iteratedQuotientMap 𝔽q β h_ℓ_add_R_rate i steps h_i_add_steps
  let k := pointToIterateQuotientIndex 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate)
    ⟨i, fin_ℓ_lt_ℓ_add_one i⟩ steps h_i_add_steps
  have hr x :=
    (is_fiber_iff_generates_quotient_point 𝔽q β i steps h_i_add_steps x (q x)).mp rfl
  have hd := Set.ncard_le_ncard_of_injOn (fun x => (q x, k x))
    (s := {x | f x ≠ g.val x}) (t := Y g ×ˢ (Set.univ : Set (Fin (2 ^ steps))))
    (fun x hx => ⟨⟨x, rfl, hx⟩, Set.mem_univ _⟩)
    (by
      intro x hx z hz h
      obtain ⟨hq, hk⟩ := Prod.mk.inj h
      exact (hr x).symm.trans ((congrArg₂ _ hq hk).trans (hr z)))
  simp only [Set.ncard_prod, Set.ncard_univ, Nat.card_fin,
    ←Finset.coe_filter_univ, Set.ncard_coe_finset] at hd
  have hb := Nat.mul_le_mul_right (2 ^ steps)
    (Nat.le_of_lt_succ (WithTop.coe_lt_coe.1 h_fw_dist_lt))
  simp only [←hg, mul_assoc, Nat.sub_mul,
    ←Nat.sub_sub,
    pow_sub_mul_pow 2 (Nat.le_sub_of_add_le' (h_i_add_steps.trans (Nat.le_add_right ℓ 𝓡))),
    pow_sub_mul_pow 2 (Nat.le_sub_of_add_le' h_i_add_steps)] at hb
  have hn := Nat.lt_succ_of_le ((Nat.mul_le_mul_left 2 hd).trans hb)
  exact (mul_le_mul_left' (Code.distFromCode_le_dist_to_mem f g.val g.property) 2).trans_lt
    (ENat.coe_lt_coe.2 hn)
