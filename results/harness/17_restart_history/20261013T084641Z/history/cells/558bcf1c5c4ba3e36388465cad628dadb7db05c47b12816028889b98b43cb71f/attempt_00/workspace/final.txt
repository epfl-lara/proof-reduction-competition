lemma time_deriv_time_deriv_electricField_of_isExtrema {A : ElectromagneticPotential d}
    {𝓕 : FreeSpace}
    (hA : ContDiff ℝ ∞ A) (J : LorentzCurrentDensity d)
    (hJ : ContDiff ℝ ∞ J) (h : IsExtrema 𝓕 A J)
    (t : Time) (x : Space d) (i : Fin d) :
    ∂ₜ (∂ₜ (A.electricField 𝓕.c · x i)) t =
      𝓕.c ^ 2 * ∑ j, (∂[j] (∂[j] (A.electricField 𝓕.c t · i)) x) -
      𝓕.c ^ 2 / 𝓕.ε₀ * ∂[i] (J.chargeDensity 𝓕.c t ·) x -
      𝓕.c ^ 2 * 𝓕.μ₀ * ∂ₜ (J.currentDensity 𝓕.c · x i) t := by
  have hA2 : ContDiff ℝ 3 A := hA.of_le ENat.LEInfty.out
  have hEs : ∀ j, ContDiff ℝ 2 (A.electricField 𝓕.c t · j) :=
    fun j => electricField_apply_contDiff_space hA2 t
  have hEd := fun j k => Space.deriv_differentiable (hEs j) k
  have hB := fun j : Fin d => magneticFieldMatrix_contDiff (n := 2) (c := 𝓕.c) A hA2 (j, i)
  have hBt := fun j => Space.space_deriv_differentiable_time (i := j) (hB j) x t
  have hJt := LorentzCurrentDensity.currentDensity_apply_differentiable_time (c := 𝓕.c)
    (hJ.differentiable (by decide)) x i t
  have hg := funext fun y =>
    ((isExtrema_iff_gauss_ampere_magneticFieldMatrix hA J hJ).mp h t y).1
  have hdiv := congrArg (fun f : Space d → ℝ => fderiv ℝ f x (Space.basis i)) hg
  unfold Space.div at hdiv
  simp only [div_eq_mul_inv] at hdiv
  rw [fderiv_fun_sum (fun j _ => (hEd j j).differentiableAt),
    fderiv_mul_const (LorentzCurrentDensity.chargeDensity_differentiable_space
      (hJ.differentiable (by decide)) t).differentiableAt] at hdiv
  simp only [_root_.sum_apply, _root_.smul_apply, smul_eq_mul,
    ← Space.deriv_eq_fderiv_basis, ← one_div] at hdiv
  conv_lhs =>
    enter [1]
    change fun s => ∂ₜ (A.electricField 𝓕.c · x i) s
    enter [s]
    rw [Time.deriv_euclid (electricField_differentiable_time (hA2.of_le (by decide)) x),
      time_deriv_electricField_of_isExtrema hA J hJ h]
  have hsum := DifferentiableAt.fun_sum (u := Finset.univ) fun j _ => hBt j
  rw [Time.deriv_eq,
    fderiv_fun_sub (hsum.const_mul _)
      (hJt.const_mul _),
    fderiv_const_mul hsum,
    fderiv_const_mul hJt,
    fderiv_fun_sum (fun j _ => hBt j)]
  simp only [_root_.sub_apply, _root_.smul_apply, _root_.sum_apply, smul_eq_mul,
    ← Time.deriv_eq]
  conv_lhs =>
    enter [1, 2, 2, j]
    rw [Space.time_deriv_comm_space_deriv (hB j)]
    simp only [time_deriv_magneticFieldMatrix A (hA2.of_le (by decide))]
    rw [Space.fun_deriv_sub _ _ (hEd i j) (hEd j i), Space.deriv_commute _ (hEs j)]
  rw [Finset.sum_sub_distrib, hdiv]
  have hc : (𝓕.c : ℝ) ^ 2 * 𝓕.μ₀ = 1 / 𝓕.ε₀ := by
    rw [FreeSpace.c_sq, ← div_div, div_mul_cancel₀ _ 𝓕.μ₀_ne_zero]
  rw [mul_comm 𝓕.μ₀ 𝓕.ε₀, ← FreeSpace.c_sq]
  simp only [mul_sub, ← mul_assoc, ← div_eq_mul_one_div, hc]
