open OldExpressions in
theorem extractedOldExprInVars :
  NormalizedOldExpr post →
  (extractOldExprVars post).Subset
  (Imperative.HasVarsPure.getVars post) := by
  have app {a b c d : List Expression.Ident} (h : a.Subset b) (h' : c.Subset d) :=
    List.append_subset.mpr ⟨List.subset_append_of_subset_left d h,
      List.subset_append_of_subset_right b h'⟩
  intro h
  induction h with
  | const | op | bvar | fvar => exact List.nil_subset _
  | abs _ ih => exact ih
  | quant _ _ ih1 ih2 | eq _ _ ih1 ih2 => exact app ih1 ih2
  | ite _ _ _ ih1 ih2 ih3 => exact app (app ih1 ih2) ih3
  | app hfn he hwf ihfn ihe =>
    unfold extractOldExprVars
    split
    · exact fun _ h => h
    · exact List.nil_subset _
    · exact app ihfn ihe
