lemma ι_timeOrderF_superCommuteF_eq_time {φ ψ : 𝓕.CrAnFieldOp}
    (hφψ : crAnTimeOrderRel φ ψ) (hψφ : crAnTimeOrderRel ψ φ) (a b : 𝓕.FieldOpFreeAlgebra) :
    ι 𝓣ᶠ(a * [ofCrAnOpF φ, ofCrAnOpF ψ]ₛF * b) =
    ι ([ofCrAnOpF φ, ofCrAnOpF ψ]ₛF * 𝓣ᶠ(a * b)) := by
  let q := [ofCrAnOpF φ, ofCrAnOpF ψ]ₛF
  let m := mulLinearMap (𝓕 := 𝓕)
  refine LinearMap.congr_fun₂
    (f := (m.comp (m.flip q)).compr₂
      (ι.toLinearMap ∘ₗ timeOrderF))
    (g := m.compr₂ (ι.toLinearMap ∘ₗ m q ∘ₗ timeOrderF))
    (ofCrAnListFBasis.ext fun φs' => ofCrAnListFBasis.ext fun φs => ?_) a b
  simp only [m, q, LinearMap.compr₂_apply, LinearMap.comp_apply, LinearMap.flip_apply,
    mulLinearMap_apply, AlgHom.toLinearMap_apply, ofListBasis_eq_ofList]
  let t := (φs' ++ φs).insertionSort crAnTimeOrderRel
  let e := fun c => crAnTimeOrderRel φ c ∧ crAnTimeOrderRel c φ
  let l1 := List.takeWhile (fun c => ¬ crAnTimeOrderRel φ c)
      t ++
    List.filter e φs'
  let l2 := List.filter e φs ++
    List.filter (fun c => crAnTimeOrderRel φ c ∧ ¬ crAnTimeOrderRel c φ)
      t
  have ht (l : List 𝓕.CrAnFieldOp)
      (h : ∀ c ∈ l, e c) :
      𝓣ᶠ(ofCrAnListF (φs' ++ l ++ φs)) =
      crAnTimeOrderSign (φs' ++ l ++ φs) •
        (ofCrAnListF l1 * ofCrAnListF l * ofCrAnListF l2) := by
    rw [timeOrderF_ofCrAnListF, crAnTimeOrderList,
      insertionSort_of_eq_list _ φ φs' l φs h]
    simp only [ofCrAnListF_append, l1, l2, t, e, mul_assoc]
  have hc := superCommuteF_ofCrAnListF_ofCrAnListF (𝓕 := 𝓕) [φ] [ψ]
  simp only [ofCrAnListF_singleton, ofList_singleton, List.singleton_append] at hc
  trans crAnTimeOrderSign (φs' ++ [φ, ψ] ++ φs) •
    (ι (ofCrAnListF l1) * ι q * ι (ofCrAnListF l2))
  · simp only [q, hc, mul_sub, sub_mul, Algebra.mul_smul_comm, Algebra.smul_mul_assoc,
      ← ofCrAnListF_append, map_sub, map_smul]
    rw [ht [φ, ψ] (by simp [e, hφψ, hψφ]), ht [ψ, φ] (by simp [e, hφψ, hψφ])]
    simp only [map_smul, map_mul, List.append_assoc, List.cons_append, List.nil_append]
    rw [crAnTimeOrderSign_swap_eq_time hφψ hψφ]
    rw [smul_sub, sub_right_inj, smul_comm]
  · have ht0 := ht [] (by simp)
    simp only [List.append_nil, List.nil_append, ofCrAnListF_nil, mul_one] at ht0
    rw [Subalgebra.mem_center_iff.mp
      (ι_superCommuteF_ofCrAnOpF_ofCrAnOpF_mem_center φ ψ), mul_assoc,
      map_mul, ← ofCrAnListF_append, ht0, map_smul, map_mul]
    rcases eq_or_ne (𝓕 |>ₛ ψ) (𝓕 |>ₛ φ) with hq | hq
    · simp only [List.append_assoc, List.cons_append, List.nil_append]
      rw [crAnTimeOrderSign,
        Wick.koszulSign_eq_rel_eq_stat _ _ hφψ hψφ hq,
        ← crAnTimeOrderSign]
      simp only [Algebra.mul_smul_comm]
    · simp only [ι_superCommuteF_of_diff_statistic (Ne.symm hq), zero_mul, smul_zero]
