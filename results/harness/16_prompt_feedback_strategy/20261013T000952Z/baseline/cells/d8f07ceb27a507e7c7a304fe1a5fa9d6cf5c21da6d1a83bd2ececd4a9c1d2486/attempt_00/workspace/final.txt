lemma ι_timeOrderF_superCommuteF_eq_time {φ ψ : 𝓕.CrAnFieldOp}
    (hφψ : crAnTimeOrderRel φ ψ) (hψφ : crAnTimeOrderRel ψ φ) (a b : 𝓕.FieldOpFreeAlgebra) :
    ι 𝓣ᶠ(a * [ofCrAnOpF φ, ofCrAnOpF ψ]ₛF * b) =
    ι ([ofCrAnOpF φ, ofCrAnOpF ψ]ₛF * 𝓣ᶠ(a * b)) := by
  let c := [ofCrAnOpF φ, ofCrAnOpF ψ]ₛF
  let T := ι.toLinearMap ∘ₗ (timeOrderF (𝓕 := 𝓕))
  let U := LinearMap.mulLeft ℂ (ι c) ∘ₗ T
  rw [map_mul ι]
  change ((mulLinearMap.comp (LinearMap.mulRight ℂ c)).compr₂ T) a b =
    (mulLinearMap.compr₂ U) a b
  refine LinearMap.congr_fun (LinearMap.congr_fun
    (ofCrAnListFBasis.ext fun φs' => ofCrAnListFBasis.ext fun φs => ?_) a) b
  change T (ofCrAnListFBasis φs' * c * ofCrAnListFBasis φs) =
    ι c * T (ofCrAnListFBasis φs' * ofCrAnListFBasis φs)
  simp only [ofListBasis_eq_ofList]
  have ht (l : List 𝓕.CrAnFieldOp) :
      T (ofCrAnListF l) = crAnTimeOrderSign l •
        ι (ofCrAnListF (crAnTimeOrderList l)) := by
    simp only [T, LinearMap.comp_apply, AlgHom.toLinearMap_apply,
      timeOrderF_ofCrAnListF, map_smul]
  let l1 := (List.takeWhile (fun c => ¬crAnTimeOrderRel φ c)
      ((φs' ++ φs).insertionSort crAnTimeOrderRel)) ++
    List.filter (fun c => crAnTimeOrderRel φ c ∧ crAnTimeOrderRel c φ) φs'
  let l2 := (List.filter (fun c => crAnTimeOrderRel φ c ∧ crAnTimeOrderRel c φ) φs) ++
    List.filter (fun c => crAnTimeOrderRel φ c ∧ ¬crAnTimeOrderRel c φ)
      ((φs' ++ φs).insertionSort crAnTimeOrderRel)
  have hs (l : List 𝓕.CrAnFieldOp)
      (h : ∀ c ∈ l, crAnTimeOrderRel φ c ∧ crAnTimeOrderRel c φ) :
      ι (ofCrAnListF (crAnTimeOrderList (φs' ++ l ++ φs))) =
        ι (ofCrAnListF l1) * ι (ofCrAnListF l) * ι (ofCrAnListF l2) := by
    rw [crAnTimeOrderList, insertionSort_of_eq_list _ φ φs' l φs h]
    simp only [l1, l2, decide_not, Bool.decide_and, ofCrAnListF_append, map_mul, mul_assoc]
  simp only [List.append_assoc, List.cons_append, List.nil_append] at hs
  have h1 := hs [φ, ψ] (by simp [hφψ, hψφ])
  have h2 := hs [ψ, φ] (by simp [hφψ, hψφ])
  simp only [List.cons_append, List.nil_append] at h1 h2
  calc
    _ = crAnTimeOrderSign (φs' ++ φ :: ψ :: φs) •
        (ι (ofCrAnListF l1) * ι c *
          ι (ofCrAnListF l2)) := by
      conv_lhs =>
        dsimp only [c]
        rw [superCommuteF_ofCrAnOpF_ofCrAnOpF]
        simp only [mul_sub, sub_mul, Algebra.mul_smul_comm, Algebra.smul_mul_assoc,
          map_sub T, map_smul T, ← ofCrAnListF_singleton, ← ofCrAnListF_append,
          List.append_assoc, List.cons_append, List.nil_append]
        rw [ht, ht]
      rw [← crAnTimeOrderSign_swap_eq_time hφψ hψφ, h1, h2]
      dsimp only [c]
      rw [superCommuteF_ofCrAnOpF_ofCrAnOpF]
      simp only [map_sub, map_smul, map_mul, ← ofCrAnListF_singleton,
        ← ofCrAnListF_append, List.cons_append, List.nil_append,
        mul_sub, sub_mul, Algebra.mul_smul_comm, Algebra.smul_mul_assoc,
        smul_sub]
      rw [smul_comm]
    _ = crAnTimeOrderSign (φs' ++ φ :: ψ :: φs) •
        (ι c *
          ι (ofCrAnListF (crAnTimeOrderList (φs' ++ φs)))) := by
      dsimp only [c]
      rw [Subalgebra.mem_center_iff.mp
        (ι_superCommuteF_ofCrAnOpF_ofCrAnOpF_mem_center φ ψ), mul_assoc]
      have h0 := hs [] (by simp)
      simp only [List.nil_append, ofCrAnListF_nil, map_one, mul_one] at h0
      rw [h0]
    _ = _ := by
      by_cases hq : (𝓕 |>ₛ φ) = (𝓕 |>ₛ ψ)
      · rw [crAnTimeOrderSign,
          Wick.koszulSign_eq_rel_eq_stat _ _ hφψ hψφ hq.symm,
          ← crAnTimeOrderSign, ← ofCrAnListF_append, ht]
        simp only [Algebra.mul_smul_comm]
      · simp only [c, ι_superCommuteF_of_diff_statistic hq, zero_mul, smul_zero]
