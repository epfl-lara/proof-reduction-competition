theorem putnam_1964_a4
(u : ℕ → ℤ)
(boundedu : ∃ B T : ℤ, ∀ n : ℕ, B ≤ u n ∧ u n ≤ T)
(hu : ∀ n ≥ 4, u n = ((u (n - 1) + u (n - 2) + u (n - 3) * u (n - 4)) : ℝ) / (u (n - 1) * u (n - 2) + u (n - 3) + u (n - 4)) ∧ (u (n - 1) * u (n - 2) + u (n - 3) + u (n - 4)) ≠ 0)
: (∃ N c : ℕ, c > 0 ∧ ∀ n ≥ N, u (n + c) = u n) := by
  obtain ⟨B, T, hb⟩ := boundedu
  obtain ⟨i, j, hij, h⟩ :=
    (Set.finite_Icc (fun _ : Fin 4 => B) (fun _ => T)).exists_lt_map_eq_of_forall_mem
      (f := fun n : ℕ => fun k : Fin 4 => u (n + k))
      (by intro n; exact ⟨fun k => (hb _).1, fun k => (hb _).2⟩)
  have hs (m : ℕ) : u (i + m) = u (j + m) := by
    induction m using Nat.strong_induction_on with
    | h m ih =>
      by_cases hm : m < 4
      · exact congrFun h ⟨m, hm⟩
      · apply Int.cast_injective (α := ℝ)
        rw [(hu (i + m) (by omega)).1, (hu (j + m) (by omega)).1]
        simp (disch := omega) only [Nat.add_sub_assoc, ih]
  refine ⟨i, j - i, Nat.sub_pos_of_lt hij, ?_⟩
  intro n hn
  simpa only [Nat.add_sub_of_le hn, ← Nat.add_sub_assoc hn,
    ← Nat.add_sub_assoc hij.le, Nat.add_comm] using (hs (n - i)).symm
