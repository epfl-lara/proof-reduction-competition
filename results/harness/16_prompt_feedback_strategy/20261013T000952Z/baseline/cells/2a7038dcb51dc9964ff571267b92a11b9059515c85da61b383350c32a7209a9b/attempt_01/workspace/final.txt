lemma fundamental_theorem_of_variational_calculus' {f : Y → V}
    (μ : Measure Y) [IsFiniteMeasureOnCompacts μ] [μ.IsOpenPosMeasure]
    [OpensMeasurableSpace Y]
    (hf : Continuous f) (hg : ∀ g, IsTestFunction g → ∫ x, ⟪f x, g x⟫_ℝ ∂μ = 0) :
    f = 0 := by
  funext x
  by_contra hx
  have hc := Continuous.inner' (𝕜 := ℝ) f (fun _ => f x) hf continuous_const
  have hp :=
    real_inner_self_nonneg'.lt_of_ne' (inner_self_eq_zero'.not.mpr hx)
  obtain ⟨r, hr, hball⟩ := Metric.isOpen_iff.mp (isOpen_lt continuous_const hc) x hp
  let φ : ContDiffBump x := ⟨r / 2, r, half_pos hr, half_lt_self hr⟩
  exact ((φ.continuous.mul hc).integral_pos_of_hasCompactSupport_nonneg_nonzero
    φ.hasCompactSupport.mul_right (fun y => φ.nonneg.eq_or_lt'.elim
      (fun hy => (mul_eq_zero_of_left hy _).ge)
      (fun hy => (mul_pos hy (hball (φ.support_eq.subset hy.ne'))).le))
    (mul_pos (φ.pos_of_mem_ball (Metric.mem_ball_self hr)) hp).ne').ne' (by
      simpa only [inner_smul_right', Pi.mul_apply] using hg _
        (IsTestFunction.smul_right ⟨φ.contDiff, φ.hasCompactSupport⟩
          (contDiff_const (c := f x))))
