theorem putnam_1964_b2
(S : Type*) [Fintype S] [Nonempty S]
(P : Finset (Set S))
(hPP : ∀ T ∈ P, ∀ U ∈ P, T ∩ U ≠ ∅)
(hPS : ¬∃ T : Set S, T ∉ P ∧ (∀ U ∈ P, T ∩ U ≠ ∅))
: (P.card = 2 ^ (Fintype.card S - 1)) := by
  classical
  simp only [ne_eq, ← Set.disjoint_iff_inter_eq_empty] at hPP hPS
  rw [← Nat.mul_left_cancel_iff Nat.two_pos,
    mul_pow_sub_one (Fintype.card_ne_zero (α := S)), ← Fintype.card_set,
    ← Set.Intersecting.is_max_iff_card_eq hPP]
  grind [Set.Intersecting]
