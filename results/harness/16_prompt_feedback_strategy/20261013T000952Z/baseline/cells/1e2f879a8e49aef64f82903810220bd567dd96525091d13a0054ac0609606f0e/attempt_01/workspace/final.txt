theorem bisimilarity_congr_choice :
    (p ~[lts (defs := defs)] q) → (choice p r) ~[lts (defs := defs)] (choice q r) := by
  intro h
  exists ChoiceBisim, ChoiceBisim.choice h
  rintro s t (h | h) μ <;>
    refine (Bisimilarity.isBisimulation _ _ h μ).imp ?_ ?_ <;> intro hf s' ht
  case choice | choice =>
    cases ht with
    | choiceL ht =>
      exact (hf _ ht).imp fun _ => And.imp Tr.choiceL ChoiceBisim.bisim
    | choiceR ht => use s', Tr.choiceR ht, ChoiceBisim.bisim (HomBisimilarity.refl _)
  all_goals exact (hf s' ht).imp fun _ => And.imp_right ChoiceBisim.bisim
