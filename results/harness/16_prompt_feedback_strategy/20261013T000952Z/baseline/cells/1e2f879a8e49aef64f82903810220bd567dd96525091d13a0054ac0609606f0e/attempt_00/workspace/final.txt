theorem bisimilarity_congr_choice :
    (p ~[lts (defs := defs)] q) → (choice p r) ~[lts (defs := defs)] (choice q r) := by
  intro h
  refine ⟨_, ChoiceBisim.choice h, ?_⟩
  intro s t hr μ
  cases hr <;> rename_i h <;>
    apply And.imp _ _ (Bisimilarity.isBisimulation _ _ h μ) <;> intro hf s' ht
  all_goals
    first
    | exact (hf s' ht).imp fun _ => And.imp_right ChoiceBisim.bisim
    | cases ht with
      | choiceL ht =>
        exact (hf _ ht).imp fun _ => And.imp Tr.choiceL ChoiceBisim.bisim
      | choiceR ht => exact ⟨s', .choiceR ht, .bisim (HomBisimilarity.refl _)⟩
