theorem bisimilarity_congr_choice :
    (p ~[lts (defs := defs)] q) → (choice p r) ~[lts (defs := defs)] (choice q r) := by
  intro h
  exists ChoiceBisim, ChoiceBisim.choice h
  rintro s t (h | h) μ <;>
    apply (Bisimilarity.isBisimulation _ _ h μ).imp <;> intro hf s' ht
  case' choice | choice => cases ht
  case choiceR | choiceR => use s', Tr.choiceR ‹_›, ChoiceBisim.bisim (HomBisimilarity.refl _)
  all_goals exact (hf _ ‹_›).imp fun _ => And.imp (by
    try exact Tr.choiceL
    all_goals exact id) ChoiceBisim.bisim
