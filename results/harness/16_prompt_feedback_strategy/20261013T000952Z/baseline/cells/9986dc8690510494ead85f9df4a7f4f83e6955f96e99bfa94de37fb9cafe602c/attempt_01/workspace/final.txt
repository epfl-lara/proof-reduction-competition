lemma Typing.progress (der : Typing [] t τ) : t.Value ∨ ∃ t', t ⭢βᵛ t' := by
  have lc := der.wf.2.1
  generalize eq : [] = Γ at der
  induction der <;> subst eq
  case app | tapp =>
    cases lc <;> grind (gen := 2) only [→ canonical_form_abs, → canonical_form_tabs,
      → Red.abs, → Red.tabs, → Red.appₗ, → Red.appᵣ, → Red.tapp]
  case let' | case =>
    grind (gen := 2) only [body_let, body_case, → canonical_form_sum, cases Value,
      → Red.let_body, → Red.let_bind, → Red.case, → Red.case_inl, → Red.case_inr]
  all_goals grind (gen := 2) [→ Red.inl, → Red.inr]
