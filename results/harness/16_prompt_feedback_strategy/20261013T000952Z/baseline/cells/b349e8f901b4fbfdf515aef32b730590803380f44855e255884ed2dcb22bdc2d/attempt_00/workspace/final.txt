theorem fiberwise_dist_lt_imp_dist_lt_unique_decoding_radius (i : Fin ℓ) (steps : ℕ)
    [NeZero steps] (h_i_add_steps : i.val + steps ≤ ℓ)
    (f : OracleFunction 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) ⟨i, by omega⟩)
  (h_fw_dist_lt : fiberwiseClose 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate)
    (i := i) (steps := steps) (h_i_add_steps := h_i_add_steps) (f := f)) :
  hammingClose 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) ⟨i, by omega⟩ f := by
  have hid : (BBF_CodeDistance ℓ 𝓡 ⟨i.val + steps, Nat.lt_succ_of_le h_i_add_steps⟩ - 1) * 2 ^ steps =
      BBF_CodeDistance ℓ 𝓡 ⟨i, fin_ℓ_lt_ℓ_add_one i⟩ - 1 := by
    dsimp only [BBF_CodeDistance, Nat.add_one_sub_one]
    rw [Nat.sub_mul, ← Nat.pow_add, ← Nat.pow_add,
      Nat.sub_add_eq_sub_sub_rev _ _ _ (Nat.le_add_left _ _) (Nat.le_add_right_of_le h_i_add_steps),
      Nat.add_sub_cancel,
      Nat.sub_add_eq_sub_sub_rev _ _ _ (Nat.le_add_left _ _) h_i_add_steps,
      Nat.add_sub_cancel]
  let C := BBF_Code 𝔽q β (ℓ := ℓ) (𝓡 := 𝓡) (h_ℓ_add_R_rate := h_ℓ_add_R_rate) ⟨i, fin_ℓ_lt_ℓ_add_one i⟩
  let D := fun g : C => (fiberwiseDisagreementSet 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) i steps h_i_add_steps f g.val).ncard
  obtain ⟨g, _, hg⟩ := Nat.sInf_mem (s := D '' Set.univ)
    (Set.image_nonempty.mpr ⟨0, Set.mem_univ _⟩)
  let Y := fiberwiseDisagreementSet 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) i steps h_i_add_steps f g.val
  let q := iteratedQuotientMap 𝔽q β h_ℓ_add_R_rate i steps h_i_add_steps
  let k := pointToIterateQuotientIndex 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) (i := ⟨i, fin_ℓ_lt_ℓ_add_one i⟩) steps h_i_add_steps
  let e : {x // f x ≠ g.val x} → Y × Fin (2 ^ steps) :=
    fun x => (⟨q x.val, ⟨x.val, rfl, x.property⟩⟩, k x.val)
  have hr (x : _) :=
    (is_fiber_iff_generates_quotient_point 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate)
      i steps h_i_add_steps x (q x)).mp rfl
  have he : Function.Injective e := by
    intro x z h
    have hq : q x.val = q z.val := congrArg (fun t => t.1.val) h
    have hk : k x.val = k z.val := congrArg Prod.snd h
    apply Subtype.ext
    exact (hr x.val).symm.trans ((congrArg₂ _ hq hk).trans (hr z.val))
  have hb := Nat.card_le_card_of_injective e he
  rw [Nat.card_prod, Nat.card_fin, Nat.card_eq_fintype_card, Fintype.card_subtype,
    Nat.card_coe_set_eq] at hb
  have hf := Nat.le_of_lt_succ (ENat.coe_lt_coe.mp h_fw_dist_lt)
  change 2 * sInf (D '' Set.univ) ≤ _ at hf
  rw [← hg] at hf
  unfold hammingClose
  apply lt_of_le_of_lt (mul_le_mul_left' (Code.distFromCode_le_dist_to_mem (C := C) f g.val g.property) 2)
  change ((2 : ℕ) : ℕ∞) * (hammingDist f g.val : ℕ∞) < _
  rw [← ENat.coe_mul, ENat.coe_lt_coe]
  apply Nat.lt_of_le_pred (Nat.succ_pos _)
  exact (Nat.mul_le_mul_left 2 hb).trans
    ((Nat.mul_assoc _ _ _).symm.le.trans ((Nat.mul_le_mul_right _ hf).trans hid.le))
