lemma time_deriv_time_deriv_electricField_of_isExtrema {A : ElectromagneticPotential d}
    {𝓕 : FreeSpace}
    (hA : ContDiff ℝ ∞ A) (J : LorentzCurrentDensity d)
    (hJ : ContDiff ℝ ∞ J) (h : IsExtrema 𝓕 A J)
    (t : Time) (x : Space d) (i : Fin d) :
    ∂ₜ (∂ₜ (A.electricField 𝓕.c · x i)) t =
      𝓕.c ^ 2 * ∑ j, (∂[j] (∂[j] (A.electricField 𝓕.c t · i)) x) -
      𝓕.c ^ 2 / 𝓕.ε₀ * ∂[i] (J.chargeDensity 𝓕.c t ·) x -
      𝓕.c ^ 2 * 𝓕.μ₀ * ∂ₜ (J.currentDensity 𝓕.c · x i) t := by
  have hA2 : ContDiff ℝ 2 A := hA.of_le ENat.LEInfty.out
  have hA3 : ContDiff ℝ 3 A := hA.of_le ENat.LEInfty.out
  have hEs j : ContDiff ℝ 2 (A.electricField 𝓕.c t · j) :=
    electricField_apply_contDiff_space hA3 t
  have hEd j k := Space.deriv_differentiable (hEs j) k
  have hBt j := Space.space_deriv_differentiable_time (i := j)
    (magneticFieldMatrix_contDiff (c := 𝓕.c) A hA3 (j, i)) x t
  have hs := DifferentiableAt.fun_sum (u := Finset.univ) (fun j _ => hBt j)
  have hJt := LorentzCurrentDensity.currentDensity_apply_differentiable_time
    (hJ.differentiable (by decide)) x i (c := 𝓕.c) t
  have hc j := Space.deriv_commute (u := j) (v := i) _ (hEs j)
  have hg := congrArg (fun f : Space d → ℝ => ∂[i] f x)
    (funext fun y => ((isExtrema_iff_gauss_ampere_magneticFieldMatrix hA J hJ).mp h t y).1)
  change ∂[i] (fun y => ∑ j, ∂[j] (A.electricField 𝓕.c t · j) y) x = _ at hg
  simp only [div_eq_inv_mul] at hg
  rw [Space.deriv_eq_fderiv_basis, fderiv_fun_sum (fun j _ => (hEd j j).differentiableAt),
    Space.deriv_eq_fderiv_basis, fderiv_const_mul
      (LorentzCurrentDensity.chargeDensity_differentiable_space
        (hJ.differentiable (by decide)) t).differentiableAt] at hg
  simp only [_root_.sum_apply, _root_.smul_apply, smul_eq_mul,
    ← Space.deriv_eq_fderiv_basis] at hg
  conv_lhs =>
    enter [1, s]
    rw [Time.deriv_euclid (electricField_differentiable_time hA2 x),
      time_deriv_electricField_of_isExtrema hA J hJ h]
  rw [Time.deriv_eq, fderiv_fun_sub
      (hs.const_mul _) (hJt.const_mul _),
    fderiv_const_mul hs, fderiv_const_mul hJt,
    fderiv_fun_sum fun j _ => hBt j]
  simp only [_root_.sub_apply, _root_.smul_apply, _root_.sum_apply, smul_eq_mul,
    ← Time.deriv_eq]
  simp_rw [Space.time_deriv_comm_space_deriv
      (magneticFieldMatrix_contDiff (c := 𝓕.c) A hA3 (_, i)),
    time_deriv_magneticFieldMatrix A hA2,
    Space.fun_deriv_sub _ _ (hEd _ _) (hEd _ _), hc]
  rw [Finset.sum_sub_distrib, hg]
  simp only [FreeSpace.c_sq, div_eq_mul_inv, mul_inv_rev, one_mul, mul_sub,
    mul_comm 𝓕.μ₀⁻¹ 𝓕.ε₀⁻¹, mul_left_comm 𝓕.μ₀⁻¹ 𝓕.ε₀⁻¹, mul_assoc, inv_mul_cancel₀ 𝓕.μ₀_ne_zero, mul_one]
