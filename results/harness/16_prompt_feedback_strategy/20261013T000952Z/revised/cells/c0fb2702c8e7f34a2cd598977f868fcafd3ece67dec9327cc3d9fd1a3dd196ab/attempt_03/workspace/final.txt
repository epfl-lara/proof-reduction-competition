theorem putnam_1964_b2
(S : Type*) [Fintype S] [Nonempty S]
(P : Finset (Set S))
(hPP : ∀ T ∈ P, ∀ U ∈ P, T ∩ U ≠ ∅)
(hPS : ¬∃ T : Set S, T ∉ P ∧ (∀ U ∈ P, T ∩ U ≠ ∅))
: (P.card = 2 ^ (Fintype.card S - 1)) := by
  apply Nat.eq_of_mul_eq_mul_left Nat.two_pos
  rw [mul_pow_sub_one Fintype.card_ne_zero, ← Fintype.card_set]
  apply (Set.Intersecting.is_max_iff_card_eq ?_).mp <;>
    simp only [Set.Intersecting, Set.disjoint_iff_inter_eq_empty, Finset.mem_coe] <;> try assumption
  clear hPP
  grind
