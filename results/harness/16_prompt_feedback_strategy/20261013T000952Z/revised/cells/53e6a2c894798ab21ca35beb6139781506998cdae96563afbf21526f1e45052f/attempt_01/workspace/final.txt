theorem putnam_1964_a4
(u : ℕ → ℤ)
(boundedu : ∃ B T : ℤ, ∀ n : ℕ, B ≤ u n ∧ u n ≤ T)
(hu : ∀ n ≥ 4, u n = ((u (n - 1) + u (n - 2) + u (n - 3) * u (n - 4)) : ℝ) / (u (n - 1) * u (n - 2) + u (n - 3) + u (n - 4)) ∧ (u (n - 1) * u (n - 2) + u (n - 3) + u (n - 4)) ≠ 0)
: (∃ N c : ℕ, c > 0 ∧ ∀ n ≥ N, u (n + c) = u n) := by
  obtain ⟨B, T, hB⟩ := boundedu
  obtain ⟨i, j, hij, he⟩ :=
    (Set.Finite.pi' fun _ : Fin 4 => Set.finite_Icc B T).exists_lt_map_eq_of_forall_mem
      (by exact fun n k => hB (n + k))
  have h m : u (i + m) = u (j + m) := Nat.strong_induction_on m fun m ih => by
    rcases lt_or_ge m 4 with hm | hm
    · exact congrFun he ⟨m, hm⟩
    · apply Int.cast_injective (α := ℝ)
      rw [(hu (i + m) (Nat.le_add_left_of_le hm)).1, (hu (j + m) (Nat.le_add_left_of_le hm)).1]
      simp (disch := omega) only [Nat.add_sub_assoc, ih]
  refine ⟨i, j - i, Nat.sub_pos_of_lt hij, ?_⟩
  intro n hn
  obtain ⟨m, rfl⟩ := Nat.exists_eq_add_of_le hn
  rw [Nat.add_right_comm, Nat.add_sub_of_le hij.le, h]
