theorem parallelReduction_diamond : Diamond ParallelReduction := by
  intro a a₁ a₂ h₁ h₂
  induction h₁ generalizing a₂
  case refl => exact ⟨_, h₂, .refl _⟩
  case' par ha hb iha ihb => skip
  all_goals cases h₂
  case par.refl | red_I.refl | red_K.refl | red_S.refl |
      red_I.red_I | red_K.red_K | red_S.red_S =>
    refine ⟨_, .refl _, ?_⟩
    constructor <;> assumption
  case par.par ha' hb' =>
    obtain ⟨_, h₃, h₄⟩ := iha ha'
    obtain ⟨_, h₅, h₆⟩ := ihb hb'
    exact ⟨_, .par h₃ h₅, .par h₄ h₆⟩
  case' red_I.par ha hb | red_K.par ha hb | red_S.par ha hb => apply symm
  case par.red_I | red_I.par =>
    cases I_irreducible _ ha
    exact ⟨_, .red_I _, hb⟩
  case par.red_K | red_K.par =>
    obtain ⟨_, h, rfl⟩ := Ka_irreducible _ _ ha
    exact ⟨_, .red_K .., h⟩
  case par.red_S | red_S.par =>
    obtain ⟨_, _, h, h', rfl⟩ := Sab_irreducible _ _ _ ha
    exact ⟨_, .red_S .., .par (.par h hb) (.par h' hb)⟩
