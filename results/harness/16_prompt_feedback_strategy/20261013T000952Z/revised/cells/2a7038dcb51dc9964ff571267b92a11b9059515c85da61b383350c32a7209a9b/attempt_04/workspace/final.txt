lemma fundamental_theorem_of_variational_calculus' {f : Y → V}
    (μ : Measure Y) [IsFiniteMeasureOnCompacts μ] [μ.IsOpenPosMeasure]
    [OpensMeasurableSpace Y]
    (hf : Continuous f) (hg : ∀ g, IsTestFunction g → ∫ x, ⟪f x, g x⟫_ℝ ∂μ = 0) :
    f = 0 := by
  funext x₀
  apply (inner_self_eq_zero' (𝕜 := ℝ)).mp
  by_contra hx₀
  have hc := Continuous.inner' (𝕜 := ℝ) f (fun _ => f x₀) hf continuous_const
  obtain ⟨φ, hs, hk, hd, hφ, hv⟩ := exists_contDiff_tsupport_subset
    (hc.continuousAt.preimage_mem_nhds (Ioi_mem_nhds (lt_of_le_of_ne' real_inner_self_nonneg' hx₀)))
  refine ((hd.continuous.fun_mul hc).integral_pos_of_hasCompactSupport_nonneg_nonzero
    hk.mul_right ?_
    (by rwa [hv, one_mul])).ne'
      (by
        simp_rw [← inner_smul_right']
        apply hg
        exact IsTestFunction.smul_right ⟨hd, hk⟩ contDiff_const)
  intro x
  by_cases hx : φ x = 0
  · simp [hx]
  · apply mul_nonneg (hφ ⟨x, rfl⟩).1
    exact le_of_lt (hs (subset_tsupport φ hx))
