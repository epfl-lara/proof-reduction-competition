theorem putnam_1995_a3
(relation : (Fin 9 → ℤ) → (Fin 9 → ℤ) → Prop)
(digits_to_num : (Fin 9 → ℤ) → ℤ)
(hdigits_to_num : digits_to_num = fun dig => ∑ i : Fin 9, (dig i) * 10^i.1)
(hrelation : ∀ d e : (Fin 9 → ℤ), relation d e ↔ (∀ i : Fin 9, d i < 10 ∧ d i ≥ 0 ∧ e i < 10 ∧ e i ≥ 0) ∧ (∀ i : Fin 9, 7 ∣ (digits_to_num (fun j : Fin 9 => if j = i then e j else d j))))
: ∀ d e f : (Fin 9 → ℤ), ((relation d e) ∧ (relation e f)) → (∀ i : Fin 9, 7 ∣ d i - f i) := by
  have step (d e : Fin 9 → ℤ) (h : relation d e) (i : Fin 9) :
      7 ∣ (e i - d i) * 10^i.1 + digits_to_num d := by
    have heq : digits_to_num (fun j => if j = i then e j else d j) =
        (e i - d i) * 10^i.1 + digits_to_num d := by
      rw [hdigits_to_num]
      have hterm (j : Fin 9) :
          (if j = i then e j else d j) * (10 : ℤ)^j.1 =
          (if j = i then (e i - d i) * 10^i.1 else 0) + d j * 10^j.1 := by
        split_ifs with hji
        · subst j; ring
        · simp
      simp_rw [hterm]
      simp [Finset.sum_add_distrib]
    rw [← heq]
    exact ((hrelation d e).mp h).2 i
  intro d e f h
  have hd := step d e h.1
  have he := step e f h.2
  have hs : 7 ∣ digits_to_num e + digits_to_num d := by
    have hh := Finset.dvd_sum (s := Finset.univ) (fun i _ => hd i)

        digits_to_num e + digits_to_num d + 7 * digits_to_num d := by
      simp only [sub_mul, Finset.sum_add_distrib, Finset.sum_sub_distrib]
      simp [hdigits_to_num, Finset.sum_const]
      ring
    rw [seq] at hh
    have := dvd_sub hh (dvd_mul_right (7 : ℤ) (digits_to_num d))
    simpa only [add_sub_cancel_right] using this

  simpa only [neg_sub] using (dvd_neg.mpr hf)
