theorem bisimilarity_congr_choice :
    (p ~[lts (defs := defs)] q) → (choice p r) ~[lts (defs := defs)] (choice q r) := by
  intro h
  refine ⟨@ChoiceBisim _ _ defs, ChoiceBisim.choice h, ?_⟩
  intro s1 s2 hr μ
  cases hr with
  | choice hb =>
    constructor
    · intro s' ht
      cases ht with
      | choiceL ht =>
        obtain ⟨t, ht, hb⟩ := hb.follow_fst ht
        exact ⟨t, Tr.choiceL ht, ChoiceBisim.bisim hb⟩
      | choiceR ht =>
        exact ⟨s', Tr.choiceR ht, ChoiceBisim.bisim (HomBisimilarity.refl _)⟩
    · intro s' ht
      cases ht with
      | choiceL ht =>
        obtain ⟨t, ht, hb⟩ := hb.follow_snd ht
        exact ⟨t, Tr.choiceL ht, ChoiceBisim.bisim hb⟩
      | choiceR ht =>
        exact ⟨s', Tr.choiceR ht, ChoiceBisim.bisim (HomBisimilarity.refl _)⟩
  | bisim hb =>
    constructor
    · intro s' ht
      obtain ⟨t, ht, hb⟩ := hb.follow_fst ht
      exact ⟨t, ht, ChoiceBisim.bisim hb⟩
    · intro s' ht
      obtain ⟨t, ht, hb⟩ := hb.follow_snd ht
      exact ⟨t, ht, ChoiceBisim.bisim hb⟩
