theorem bisimilarity_congr_choice :
    (p ~[lts (defs := defs)] q) → (choice p r) ~[lts (defs := defs)] (choice q r) := by
  intro h
  refine ⟨@ChoiceBisim _ _ defs, .choice h, ?_⟩
  intro s1 s2 hr μ
  cases hr with
  | choice h =>
    constructor <;> intro s' ht
    · cases ht with
      | choiceL ht =>
        obtain ⟨t', ht', h'⟩ := h.follow_fst ht
        exact ⟨t', Tr.choiceL ht', .bisim h'⟩
      | choiceR ht =>
        exact ⟨s', Tr.choiceR ht, .bisim (HomBisimilarity.refl _)⟩
    · cases ht with
      | choiceL ht =>
        obtain ⟨t', ht', h'⟩ := h.follow_snd ht
        exact ⟨t', Tr.choiceL ht', .bisim h'⟩
      | choiceR ht =>
        exact ⟨s', Tr.choiceR ht, .bisim (HomBisimilarity.refl _)⟩
  | bisim h =>
    constructor <;> intro s' ht
    · obtain ⟨t', ht', h'⟩ := h.follow_fst ht
      exact ⟨t', ht', .bisim h'⟩
    · obtain ⟨t', ht', h'⟩ := h.follow_snd ht
      exact ⟨t', ht', .bisim h'⟩
