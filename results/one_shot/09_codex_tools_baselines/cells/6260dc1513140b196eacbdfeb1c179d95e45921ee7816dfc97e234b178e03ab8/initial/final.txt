theorem fiberwise_dist_lt_imp_dist_lt_unique_decoding_radius (i : Fin ℓ) (steps : ℕ)
    [NeZero steps] (h_i_add_steps : i.val + steps ≤ ℓ)
    (f : OracleFunction 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) ⟨i, by omega⟩)
  (h_fw_dist_lt : fiberwiseClose 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate)
    (i := i) (steps := steps) (h_i_add_steps := h_i_add_steps) (f := f)) :
  hammingClose 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) ⟨i, by omega⟩ f := by
  classical
  let C := BBF_Code 𝔽q β (h_ℓ_add_R_rate := h_ℓ_add_R_rate) ⟨i, by omega⟩
  let D := fun g : C => (fiberwiseDisagreementSet 𝔽q β i steps h_i_add_steps f g).ncard
  obtain ⟨g, _, hg⟩ := Nat.sInf_mem
    (show (D '' Set.univ).Nonempty from Set.image_nonempty.mpr Set.univ_nonempty)
  let Y := fiberwiseDisagreementSet 𝔽q β i steps h_i_add_steps f g
  have hfw : 2 * Y.ncard ≤
      2 ^ (ℓ + 𝓡 - (i.val + steps)) - 2 ^ (ℓ - (i.val + steps)) := by
    apply Nat.le_of_lt_succ
    apply ENat.coe_lt_coe.mp
    change 2 * (D g : ℕ∞) < (BBF_CodeDistance ℓ 𝓡 ⟨i.val + steps, by omega⟩ : ℕ∞)
    rw [hg]
    exact h_fw_dist_lt
  let fibers := fun y : sDomain 𝔽q β h_ℓ_add_R_rate ⟨i.val + steps, by omega⟩ =>
    Finset.univ.image
      (qMap_total_fiber 𝔽q β (i := ⟨i, by omega⟩) (steps := steps)
        (h_i_add_steps := fin_ℓ_steps_lt_ℓ_add_R i steps h_i_add_steps) y)
  have hdist : hammingDist f g ≤ Y.ncard * 2 ^ steps := by
    rw [Set.ncard_eq_toFinset_card']
    apply le_trans (b := (Y.toFinset.biUnion fibers).card)
    · apply Finset.card_le_card
      intro x hx
      refine Finset.mem_biUnion.mpr
        ⟨iteratedQuotientMap 𝔽q β h_ℓ_add_R_rate i steps h_i_add_steps x, ?_, ?_⟩
      · exact Set.mem_toFinset.mpr ⟨x, rfl, (Finset.mem_filter.mp hx).2⟩
      · exact Finset.mem_image.mpr ⟨_, Finset.mem_univ _,
          (is_fiber_iff_generates_quotient_point 𝔽q β i steps h_i_add_steps x _).mp rfl⟩
    · apply Finset.card_biUnion_le_card_mul
      intro y hy
      exact Finset.card_image_le.trans_eq (Finset.card_fin _)
  have hmul : 2 * hammingDist f g ≤ 2 ^ (ℓ + 𝓡 - i.val) - 2 ^ (ℓ - i.val) := by
    calc
      _ ≤ 2 * (Y.ncard * 2 ^ steps) := Nat.mul_le_mul_left 2 hdist
      _ = (2 * Y.ncard) * 2 ^ steps := (Nat.mul_assoc _ _ _).symm
      _ ≤ (2 ^ (ℓ + 𝓡 - (i.val + steps)) - 2 ^ (ℓ - (i.val + steps))) * 2 ^ steps :=
        Nat.mul_le_mul_right _ hfw
      _ = _ := by
        rw [Nat.sub_mul, ← Nat.pow_add, ← Nat.pow_add,
          show ℓ + 𝓡 - (i.val + steps) + steps = ℓ + 𝓡 - i.val by omega,
          show ℓ - (i.val + steps) + steps = ℓ - i.val by omega]
  unfold hammingClose
  apply lt_of_le_of_lt (mul_le_mul_left' (Code.distFromCode_le_dist_to_mem f g g.property) 2)
  exact ENat.coe_lt_coe.mpr (Nat.lt_succ_of_le hmul)
