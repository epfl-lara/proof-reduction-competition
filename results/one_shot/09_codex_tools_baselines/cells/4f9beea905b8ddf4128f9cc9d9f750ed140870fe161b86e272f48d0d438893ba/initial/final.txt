theorem parallelReduction_diamond : Diamond ParallelReduction := by
  intro a a₁ a₂ h₁ h₂
  induction h₁ generalizing a₂
  case refl => exact ⟨_, h₂, .refl _⟩
  case red_I a =>
    cases h₂
    case refl => exact ⟨_, .refl _, .red_I _⟩
    case red_I => exact ⟨_, .refl _, .refl _⟩
    case par c a' hc ha =>
      rw [I_irreducible _ hc]
      exact ⟨_, ha, .red_I _⟩
  case red_K a b =>
    cases h₂
    case refl => exact ⟨_, .refl _, .red_K _ _⟩
    case red_K => exact ⟨_, .refl _, .refl _⟩
    case par a' b' ha hb =>
      obtain ⟨u, hu, rfl⟩ := Ka_irreducible _ _ ha
      exact ⟨_, hu, .red_K _ _⟩
  case red_S a b c =>
    cases h₂
    case refl => exact ⟨_, .refl _, .red_S _ _ _⟩
    case red_S => exact ⟨_, .refl _, .refl _⟩
    case par d c' hd hc =>
      obtain ⟨u, v, hu, hv, rfl⟩ := Sab_irreducible _ _ _ hd
      exact ⟨_, .par (.par hu hc) (.par hv hc), .red_S _ _ _⟩
  case par a a' b b' ha hb iha ihb =>
    cases h₂
    case refl => exact ⟨_, .refl _, .par ha hb⟩
    case red_I =>
      rw [I_irreducible _ ha]
      exact ⟨_, .red_I _, hb⟩
    case red_K =>
      obtain ⟨u, hu, rfl⟩ := Ka_irreducible _ _ ha
      exact ⟨_, .red_K _ _, hu⟩
    case red_S =>
      obtain ⟨u, v, hu, hv, rfl⟩ := Sab_irreducible _ _ _ ha
      exact ⟨_, .red_S _ _ _, .par (.par hu hb) (.par hv hb)⟩
    case par a'' b'' ha'' hb'' =>
      obtain ⟨u, hu, hu'⟩ := iha ha''
      obtain ⟨v, hv, hv'⟩ := ihb hb''
      exact ⟨_, .par hu hv, .par hu' hv'⟩
