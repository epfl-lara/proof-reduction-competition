theorem putnam_1995_a3
(relation : (Fin 9 → ℤ) → (Fin 9 → ℤ) → Prop)
(digits_to_num : (Fin 9 → ℤ) → ℤ)
(hdigits_to_num : digits_to_num = fun dig => ∑ i : Fin 9, (dig i) * 10^i.1)
(hrelation : ∀ d e : (Fin 9 → ℤ), relation d e ↔ (∀ i : Fin 9, d i < 10 ∧ d i ≥ 0 ∧ e i < 10 ∧ e i ≥ 0) ∧ (∀ i : Fin 9, 7 ∣ (digits_to_num (fun j : Fin 9 => if j = i then e j else d j))))
: ∀ d e f : (Fin 9 → ℤ), ((relation d e) ∧ (relation e f)) → (∀ i : Fin 9, 7 ∣ d i - f i) := by
  have step (a b : Fin 9 → ℤ) (hab : relation a b) (i : Fin 9) :
      7 ∣ (b i - a i) * 10 ^ i.1 + digits_to_num a := by
    convert ((hrelation a b).mp hab).2 i using 1
    rw [hdigits_to_num]
    calc
      _ = ∑ j : Fin 9, ((if j = i then (b j - a j) * 10 ^ j.1 else 0) +
          a j * 10 ^ j.1) := by
        rw [Finset.sum_add_distrib]
        simp only [Finset.sum_ite_eq', Finset.mem_univ, if_true]
      _ = _ := by
        apply Finset.sum_congr rfl
        intro j _
        by_cases hij : j = i <;> simp only [hij, ↓reduceIte] <;> ring
  intro d e f h
  have hd := step d e h.1
  have he := step e f h.2
  have hs : 7 ∣ digits_to_num e - digits_to_num d + 9 * digits_to_num d := by
    have hs := Finset.dvd_sum (s := Finset.univ) (fun i _ => hd i)
    simpa only [hdigits_to_num, sub_mul, Finset.sum_add_distrib,
      Finset.sum_sub_distrib, Finset.sum_const, Finset.card_univ,
      Fintype.card_fin, Int.nsmul_eq_mul, Nat.cast_ofNat] using hs
  have hsum : 7 ∣ digits_to_num d + digits_to_num e := by
    convert dvd_sub hs (dvd_mul_right (7 : ℤ) (digits_to_num d)) using 1 <;> ring
  intro i
  have hi : 7 ∣ (d i - f i) * (10 : ℤ) ^ i.1 := by
    convert dvd_sub hsum (dvd_add (hd i) (he i)) using 1 <;> ring
  have hp : Prime (7 : ℤ) := Int.prime_iff_natAbs_prime.mpr (by decide)
  exact (hp.dvd_mul.mp hi).resolve_right
    (fun h => (by decide : ¬ (7 : ℤ) ∣ 10) (hp.dvd_of_dvd_pow h))
